## Review Focus

1. A boolean column carrying a **bare ISON boolean** (`true` unquoted) instead of the `"true"` token — must be `MALFORMED_DATAMATICS`, never accepted as `True` (Task 3 `test_bare_boolean_rejected`).
2. A node row with an **`x-` value in a closed non-role column** (`emphasis x-huge`) — must be `MALFORMED_DATAMATICS`: the extension namespace applies only to `role`, `ring`, `a11y_role`, `shape_role`, `pattern_role`, `family` (Task 3 `test_x_namespace_only_where_allowed`).
3. A node that is a member of a `collapse` group **and** an `aggregate` group — must be `MALFORMED_DATAMATICS` (exclusivity spans both intents) (Task 4 `test_collapse_aggregate_exclusivity`).
4. A **long parent chain** that exceeds `max_group_depth` but has no cycle — `DATAMATICS_LIMIT_EXCEEDED` at D3, never a recursion error (Task 4 `test_group_depth_budget_iterative`).
5. A member whose node is **absent from the target** while a node-hint row for a *different* node is present — the failure names the member (`table.members.<g>/<type>:<id>`), not the node row (Task 5 `test_unresolved_member_names_member`).

from __future__ import annotations


class IGDMError(Exception):
    def __init__(self, code: str, field: str, detail: str, step: str) -> None:
        super().__init__(f"{step} {code} {field}: {detail}")
        self.code = code
        self.field = field
        self.detail = detail
        self.step = step

# isongraph-ext/tests/test_igdm_payload.py
import pytest
from igdm_helpers import REFERENCE_TEXT, SHA_T, payload

from isongraph_ext.igdm.errors import IGDMError
from isongraph_ext.igdm.payload import DatamaticsPayload
from isongraph_ext.igdm.records import EdgeHints, Group, LayoutHints, Limits, Member, NodeHints
from isongraph_ext.igpv.records import EdgeAddress, NodeRef
from isongraph_ext.igx.digest import sha256_hex


def test_to_ison_matches_spec_literal_oracle():
    assert payload().to_ison() == REFERENCE_TEXT


def test_roundtrip_and_digest():
    p = payload()
    text = p.to_ison()
    q = DatamaticsPayload.from_ison(text)
    assert q == p and q.to_ison() == text and p.digest == sha256_hex(text.encode("utf-8"))
    assert DatamaticsPayload.from_ison(text.encode("utf-8")) == p


def _rejects(text, field=None, code="MALFORMED_DATAMATICS", limits=None):
    with pytest.raises(IGDMError) as info:
        DatamaticsPayload.from_ison(text, **({"limits": limits} if limits else {}))
    assert (info.value.code, info.value.step) == (code, "D1"), info.value
    if field:
        assert info.value.field == field, info.value
    return info.value


def test_canonical_form_and_blocks():
    text = payload().to_ison()
    _rejects(text + "\n", "canonical_form")
    _rejects(text.replace("\n", "\r\n"), "canonical_form")
    _rejects(text.replace("cluster 1 order\nregion 1 order", "region 1 order\ncluster 1 order"), "canonical_form")
    _rejects(text + "\n\nmeta.extra\na\n1", "meta.extra")
    _rejects(text.replace("IGDM/1", "IGDM/2"), "meta.igdm.datamatics_spec")
    _rejects(text.replace("table.members", "table.memberz"), "table.memberz")
    _rejects(b"\xff", "document")


def test_bare_boolean_rejected():
    # Review Focus 1: booleans are "true"/"false" string tokens; a bare ISON boolean is malformed, never coerced.
    text = payload().to_ison()
    _rejects(text.replace('"true" document', "true document"), "table.nodes.order:1.selectable")
    _rejects(text.replace('null "false" region', "null false region"), "table.groups.region.collapsed")


def test_x_namespace_only_where_allowed():
    # Review Focus 2: x- values are legal in role-like columns only.
    def bad(field, **kw):

"""IGDM/1 payload: wire model (19 §1), D1 field rules (check). Task 4 adds relation rules, check_references, unrecognized_roles.

def _err(field: str, detail: str) -> IGDMError:
    return IGDMError("MALFORMED_DATAMATICS", field, detail, "D1")


def _limit(name: str, detail: str) -> IGDMError:
    return IGDMError("DATAMATICS_LIMIT_EXCEEDED", name, detail, "D1")


def node_key(ref: PR.NodeRef) -> str:
    return f"{ref.node_type}:{ref.node_id}"


def edge_key(e: PR.EdgeAddress) -> str:
    return f"{node_key(e.source)}-{e.relation}-{node_key(e.target)}#{e.ordinal}"


def _tok(v: bool | None) -> str | None:
    return None if v is None else ("true" if v else "false")


def _node_row(n: R.NodeHints) -> dict:
    row = {"node_type": n.node.node_type, "node_id": n.node.node_id}
    for col in R.NODE_HINTS:
        v = getattr(n, col)
        row[col] = _tok(v) if col in R.BOOL_COLUMNS else v
    return row


def _edge_row(e: R.EdgeHints) -> dict:
    a = e.edge
    row = {"source_type": a.source.node_type, "source_id": a.source.node_id, "relation": a.relation,
           "target_type": a.target.node_type, "target_id": a.target.node_id, "ordinal": a.ordinal}
    for col in R.EDGE_HINTS:
        v = getattr(e, col)
        row[col] = _tok(v) if col in R.BOOL_COLUMNS else v
    return row


def _group_row(g: R.Group) -> dict:
    return {col: (_tok(getattr(g, col)) if col in R.BOOL_COLUMNS else getattr(g, col)) for col in R.GROUP_FIELDS}


def _member_row(m: R.Member) -> dict:
    return {"group_id": m.group_id, "node_type": m.node.node_type, "node_id": m.node.node_id}


def _untok(v: Any, where: str) -> bool | None:
    if v is None:
        return None
    if v == "true" and isinstance(v, str):
        return True
    if v == "false" and isinstance(v, str):
        return False
    raise _err(where, 'boolean columns carry the string tokens "true" / "false"')


def _node_from_row(r: dict) -> R.NodeHints:
    ref = PR.NodeRef(r["node_type"], r["node_id"])
    kw = {col: (_untok(r[col], f"table.nodes.{node_key(ref)}.{col}") if col in R.BOOL_COLUMNS else r[col]) for col in R.NODE_HINTS}
    return R.NodeHints(ref, **kw)


def _edge_from_row(r: dict) -> R.EdgeHints:
    addr = PR.EdgeAddress(PR.NodeRef(r["source_type"], r["source_id"]), r["relation"], PR.NodeRef(r["target_type"], r["target_id"]), r["ordinal"])
    kw = {col: (_untok(r[col], f"table.edges.{edge_key(addr)}.{col}") if col in R.BOOL_COLUMNS else r[col]) for col in R.EDGE_HINTS}
    return R.EdgeHints(addr, **kw)


def _group_from_row(r: dict) -> R.Group:
    kw = {col: (_untok(r[col], f"table.groups.{r['group_id']}.{col}") if col in R.BOOL_COLUMNS else r[col]) for col in R.GROUP_FIELDS}
    return R.Group(**kw)


@dataclass(frozen=True)
class DatamaticsPayload:
    target_artifact_sha256: str
    target_serialization_profile: str
    layout: R.LayoutHints = R.LayoutHints()
    nodes: tuple[R.NodeHints, ...] = ()
    edges: tuple[R.EdgeHints, ...] = ()
    groups: tuple[R.Group, ...] = ()
    members: tuple[R.Member, ...] = ()

    def __post_init__(self) -> None:
        # Records are held in a stable order so value equality is independent of construction/wire order (as 18).
        def norm(rows, key):
            try:
                return tuple(sorted(rows, key=lambda r: _key_bytes(key(r))))
            except Exception:
                return tuple(rows)
        object.__setattr__(self, "nodes", norm(self.nodes, lambda n: node_key(n.node)))
        object.__setattr__(self, "edges", norm(self.edges, lambda e: edge_key(e.edge)))
        object.__setattr__(self, "groups", norm(self.groups, lambda g: g.group_id))
        object.__setattr__(self, "members", norm(self.members, lambda m: f"{m.group_id}/{node_key(m.node)}"))

    def to_document(self) -> Document:
        return Document(blocks=[
            Block("meta", "igdm", list(R.META_FIELDS), [{"datamatics_spec": R.DATAMATICS_SPEC, "target_artifact_sha256": self.target_artifact_sha256,
                                                          "target_serialization_profile": self.target_serialization_profile}]),
            Block("meta", "layout", list(R.LAYOUT_FIELDS), [{"family": self.layout.family, "direction": self.layout.direction, "stability": self.layout.stability}]),
            Block("table", "nodes", list(R.NODE_FIELDS), [_node_row(n) for n in self.nodes]),
            Block("table", "edges", list(R.EDGE_FIELDS), [_edge_row(e) for e in self.edges]),
            Block("table", "groups", list(R.GROUP_FIELDS), [_group_row(g) for g in self.groups]),
            Block("table", "members", list(R.MEMBER_FIELDS), [_member_row(m) for m in self.members]),
        ])

    def to_ison(self) -> str:
        self.check()
        return dumps_canonical(self.to_document())

    @property
    def digest(self) -> str:
        return sha256_hex(self.to_ison())

    @classmethod
    def from_ison(cls, text: str | bytes, *, limits: R.Limits = R.DEFAULT_LIMITS) -> "DatamaticsPayload":
        if isinstance(text, bytes):
            data = text
            try:
                text = data.decode("utf-8")
            except UnicodeDecodeError as e:
                raise _err("document", f"not UTF-8: {e}") from None
        else:
            try:
                data = text.encode("utf-8")
            except UnicodeEncodeError as e:
                raise _err("document", f"text is not UTF-8-encodable: {e}") from None
        if len(data) > limits.max_payload_bytes:
            raise _limit("max_payload_bytes", f"{len(data)} > {limits.max_payload_bytes}")
        if "\r" in text:
            raise _err("canonical_form", "CR bytes present; the canonical writer emits LF only")
        try:
            doc = loads(text)
        except Exception as e:
            raise _err("document", f"not ISON: {e}") from None
        blocks = _collect_blocks_for(doc, REQUIRED_BLOCKS, _err)
        meta = _one_row(blocks[("meta", "igdm")], "meta.igdm", R.META_FIELDS, _err)
        if meta["datamatics_spec"] != R.DATAMATICS_SPEC:
            raise _err("meta.igdm.datamatics_spec", f"must be {R.DATAMATICS_SPEC}")
        lay = _one_row(blocks[("meta", "layout")], "meta.layout", R.LAYOUT_FIELDS, _err)
        rows: dict[str, list[dict]] = {}
        for name, fields in TABLES:
            b = blocks[("table", name)]
            if set(b.fields) != set(fields):
                raise _err(f"table.{name}", f"fields {sorted(b.fields)} != {sorted(fields)}")
            if len(b.rows) > limits.max_rows_per_table:
                raise _limit("max_rows_per_table", f"table.{name} has {len(b.rows)} rows > {limits.max_rows_per_table}")
            rows[name] = b.rows
        p = cls(target_artifact_sha256=meta["target_artifact_sha256"], target_serialization_profile=meta["target_serialization_profile"],
                layout=R.LayoutHints(lay["family"], lay["direction"], lay["stability"]),
                nodes=tuple(_node_from_row(r) for r in rows["nodes"]), edges=tuple(_edge_from_row(r) for r in rows["edges"]),
                groups=tuple(_group_from_row(r) for r in rows["groups"]),
                members=tuple(R.Member(r["group_id"], PR.NodeRef(r["node_type"], r["node_id"])) for r in rows["members"]))
        if p.to_ison() != text:
            raise _err("canonical_form", "payload bytes are not the canonical writer's output for this document (row order included)")
        return p

    def check(self) -> None:
        """D1 rules that need no bytes (19 §3.1): field rules, then relation rules."""
        _check_fields(self)
        _check_relations(self)


def _need(cond: bool, field: str, detail: str) -> None:

def _vocab(v: Any, col: str, where: str) -> None:

def _ancestry(groups: dict[str, R.Group]) -> tuple[dict[str, frozenset[str]], dict[str, int]]:
    """Ancestor set and depth of every group, computed ONCE with memoised iterative walks: O(G * h) total, never per pair.
    Unknown parents and cycles end a walk (D3 reports them); D1 only needs 'who is above whom'."""
    anc: dict[str, frozenset[str]] = {}
    depth: dict[str, int] = {}
    for start in groups:
        if start in anc:
            continue
        path: list[str] = []
        cur: str | None = start
        seen: set[str] = set()
        while cur is not None and cur in groups and cur not in anc and cur not in seen:
            path.append(cur)
            seen.add(cur)
            cur = groups[cur].parent_group_id
        base: frozenset[str] = anc.get(cur, frozenset()) if cur is not None and cur in anc else frozenset()
        base_depth = depth.get(cur, 0) if cur is not None and cur in anc else 0
        for gid in reversed(path):                       # from the top of the walk down: ancestors of gid are everything above it
            anc[gid] = base
            depth[gid] = base_depth + 1
            base = base | {gid}
            base_depth += 1
    return anc, depth


def _check_relations(p: DatamaticsPayload) -> None:
    _twin_and_unique(p)
    groups = {g.group_id: g for g in p.groups}
    anc, depth = _ancestry(groups)
    by_node: dict[PR.NodeRef, list[str]] = {}
    for m in p.members:                                  # members are in key order (group_id first), so conflicts report the later one
        by_node.setdefault(m.node, []).append(m.group_id)
    for node, gids in by_node.items():
        excl = [g for g in gids if g in groups and groups[g].intent in ("collapse", "aggregate")]
        if len(excl) > 1:
            _need(False, f"table.members.{excl[1]}/{node_key(node)}.group_id", f"node already belongs to collapse/aggregate group {excl[0]!r}")
        hier = sorted((g for g in gids if g in groups and groups[g].intent == "hierarchy"), key=lambda g: depth[g])
        for shallow, deep in zip(hier, hier[1:]):       # a chain iff each next-deeper group has the previous one as an ancestor
            _need(shallow in anc[deep], f"table.members.{deep}/{node_key(node)}.group_id", f"hierarchy groups {shallow!r} and {deep!r} are not nested")
    with_members = {gid for gids in by_node.values() for gid in gids if gid in groups}
    reachable: set[str] = set(with_members)
    for gid in with_members:
        reachable.update(anc[gid])
    for g in p.groups:
        _need(g.group_id in reachable, f"table.groups.{g.group_id}.group_id", "unreachable: no members and no descendant with members")


def check_references(p: DatamaticsPayload, limits: R.Limits = R.DEFAULT_LIMITS) -> None:
    """D3 (19 §3.1): parents and member groups exist; the parent relation is acyclic and within max_group_depth."""
    groups = {g.group_id: g for g in p.groups}
    for g in p.groups:
        if g.parent_group_id is not None and g.parent_group_id not in groups:
            raise IGDMError("UNRESOLVED_REFERENCE", f"table.groups.{g.group_id}.parent_group_id", f"unknown group {g.parent_group_id!r}", "D3")
    for m in p.members:
        if m.group_id not in groups:
            raise IGDMError("UNRESOLVED_REFERENCE", f"table.members.{m.group_id}/{node_key(m.node)}.group_id", f"unknown group {m.group_id!r}", "D3")
    for g in p.groups:                      # iterative walk: a 10^4-deep chain must never recurse
        seen = {g.group_id}
        depth = 0
        cur = g.parent_group_id
        while cur is not None:
            if cur in seen:
                raise IGDMError("GROUP_CYCLE", f"table.groups.{g.group_id}.parent_group_id", f"group {cur!r} is its own ancestor", "D3")
            depth += 1
            if depth > limits.max_group_depth:
                raise IGDMError("DATAMATICS_LIMIT_EXCEEDED", "max_group_depth", f"group {g.group_id!r} nests deeper than {limits.max_group_depth}", "D3")
            seen.add(cur)
            cur = groups[cur].parent_group_id

    """igpv.resolve raises IGPVError with step P4; re-badge it as an IGDMError at D4 with the IGDM row field."""
    col = getattr(e, "field", where).rsplit(".", 1)[-1]
    return IGDMError("UNRESOLVED_OBJECT", f"{where}.{col}", getattr(e, "detail", str(e)), "D4")


def _resolve(graph, obj: GraphObject, where: str) -> None:
    try:
        _igpv_resolve_object(graph, obj, where)
    except Exception as e:
        raise _p4_error(e, where) from None


def verify(payload: str | bytes | DatamaticsPayload, binding: ExtensionBinding, target: str | bytes, *,
           limits: R.Limits = R.DEFAULT_LIMITS, baselines: Mapping[str, Baseline] = BASELINES,
           registry: CapabilityRegistry = DEFAULT_REGISTRY) -> tuple[VerificationResult, DatamaticsVerification]:
    target_bytes = as_bytes(target)
    if len(target_bytes) > limits.max_graph_bytes:                                            # preflight
        raise _limit("max_graph_bytes", f"target {len(target_bytes)} > {limits.max_graph_bytes}")
    payload_bytes = _payload_bytes(payload)
    if len(payload_bytes) > limits.max_payload_bytes:
        raise _limit("max_payload_bytes", f"payload {len(payload_bytes)} > {limits.max_payload_bytes}")
    igx_result = igx_verify(binding, target_bytes, baselines=baselines, registry=registry)   # D0
    row = next((c for c in binding.capabilities if c.extension == R.CAPABILITY), None)
    if row is None or row.payload_sha256 is None:
        raise IGDMError("DATAMATICS_NOT_BOUND", "table.capabilities.igdm", "the binding declares no igdm payload digest", "D0")
    if not range_admits(parse_range(row.version_range), parse_version(R.CAPABILITY_VERSION)):
        raise IGDMError("DATAMATICS_VERSION_UNSUPPORTED", "table.capabilities.igdm.version_range", f"{row.version_range} excludes IGDM/1", "D0")
    actual = sha256_hex(payload_bytes)
    if actual != row.payload_sha256:
        raise IGDMError("PAYLOAD_DIGEST_MISMATCH", "table.capabilities.igdm.payload_sha256", f"payload bytes hash to {actual}", "D0")
    if isinstance(payload, DatamaticsPayload):                                                 # D1
        p = payload
        for name in ("nodes", "edges", "groups", "members"):
            if len(getattr(p, name)) > limits.max_rows_per_table:
                raise _limit("max_rows_per_table", f"table.{name} has {len(getattr(p, name))} rows > {limits.max_rows_per_table}")
    else:
        p = DatamaticsPayload.from_ison(payload_bytes, limits=limits)
    t_actual = sha256_hex(target_bytes)                                                        # D2
    if p.target_artifact_sha256 != t_actual:
        raise IGDMError("TARGET_BINDING_MISMATCH", "meta.igdm.target_artifact_sha256", f"target bytes hash to {t_actual}", "D2")
    if p.target_artifact_sha256 != binding.artifact_sha256:
        raise IGDMError("TARGET_BINDING_MISMATCH", "meta.igdm.target_artifact_sha256", f"binding names {binding.artifact_sha256}", "D2")
    if p.target_serialization_profile != binding.serialization_profile:
        raise IGDMError("TARGET_BINDING_MISMATCH", "meta.igdm.target_serialization_profile", f"{p.target_serialization_profile} != binding {binding.serialization_profile}", "D2")
    check_references(p, limits)                                                                # D3
    if igx_result.semantic_verification == "full":                                            # D4
        try:
            graph = _igpv_load_graph(target_bytes, binding.serialization_profile, "target")
        except Exception as e:
            raise IGDMError("UNRESOLVED_OBJECT", "target", getattr(e, "detail", str(e)), "D4") from None
        for n in p.nodes:
            _resolve(graph, GraphObject("n", "node", node=n.node), f"table.nodes.{node_key(n.node)}")
        for e in p.edges:
            _resolve(graph, GraphObject("e", "edge", edge=e.edge), f"table.edges.{edge_key(e.edge)}")
        for m in p.members:
            _resolve(graph, GraphObject("m", "node", node=m.node), f"table.members.{m.group_id}/{node_key(m.node)}")
        object_resolution = "full"
    else:
        object_resolution = "skipped"
    counts = {name: len(getattr(p, name)) for name in ("nodes", "edges", "groups", "members")}
    return igx_result, DatamaticsVerification(payload_sha256=actual, binding_digest=binding.digest, target_artifact_sha256=p.target_artifact_sha256,
                                              object_resolution=object_resolution, unrecognized_roles=unrecognized_roles(p),
                                              counts=counts, steps_passed=STEPS)

def bind(target: str | bytes, payload: DatamaticsPayload, *, requirement: str = "REQUIRED", serialization_profile: str,
         producer_port: str, identity_claim: str, capabilities: Sequence[Capability] = (), consumer: Consumer | None = None,
         baseline: Baseline = DEFAULT_BASELINE, registry: CapabilityRegistry = DEFAULT_REGISTRY,
         policy: AdmissionPolicy = DEFAULT_POLICY, limits: R.Limits = R.DEFAULT_LIMITS) -> ExtensionBinding:
    if requirement not in ("REQUIRED", "OPTIONAL"):
        raise IGDMError("MALFORMED_DATAMATICS", "table.capabilities.igdm.requirement", "REQUIRED or OPTIONAL", "D1")
    if any(c.extension == R.CAPABILITY for c in capabilities):
        raise IGDMError("MALFORMED_DATAMATICS", "table.capabilities.igdm", "bind() adds the igdm row itself; do not pass one", "D1")
    payload.check()
    data = as_bytes(target)
    actual = sha256_hex(data)
    if payload.target_artifact_sha256 != actual:
        raise IGDMError("TARGET_BINDING_MISMATCH", "meta.igdm.target_artifact_sha256", f"target bytes hash to {actual}", "D2")
    if payload.target_serialization_profile != serialization_profile:
        raise IGDMError("TARGET_BINDING_MISMATCH", "meta.igdm.target_serialization_profile", f"{payload.target_serialization_profile} != {serialization_profile}", "D2")
    caps = tuple(capabilities) + (Capability(R.CAPABILITY, R.CAPABILITY_RANGE, requirement, payload.digest),)
    binding = igx_bind(data, serialization_profile=serialization_profile, producer_port=producer_port, identity_claim=identity_claim,
                       capabilities=caps, consumer=consumer, baseline=baseline, registry=registry, policy=policy)
    verify(payload, binding, data, limits=limits, baselines={**BASELINES, baseline.core_commit: baseline}, registry=registry)
    return binding


def verify_and_admit(payload: str | bytes | DatamaticsPayload, binding: ExtensionBinding, target: str | bytes, *,
                     limits: R.Limits = R.DEFAULT_LIMITS, baselines: Mapping[str, Baseline] = BASELINES,
                     registry: CapabilityRegistry = DEFAULT_REGISTRY, policy: AdmissionPolicy = DEFAULT_POLICY
                     ) -> tuple[VerificationResult, DatamaticsVerification, AdmissionResult]:
    result, dm = verify(payload, binding, target, limits=limits, baselines=baselines, registry=registry)
    return result, dm, admit(binding, result, policy=policy, baselines=baselines)

"""CLI (19 §4): verify -> 0/1/2, 64 on usage; one canonical ISON document per response."""

from __future__ import annotations

import argparse
import sys
from collections.abc import Mapping
from pathlib import Path

from ..igx.binding import ExtensionBinding

from .errors import IGDMError

EX_USAGE = 64


class _Parser(argparse.ArgumentParser):
    def error(self, message: str) -> None:  # type: ignore[override]
        self.print_usage(sys.stderr)
        sys.stderr.write(f"error: {message}\n")
        raise SystemExit(EX_USAGE)


def _utf8_streams() -> None:
    for stream in (sys.stdout, sys.stderr):
        reconfigure = getattr(stream, "reconfigure", None)
        if reconfigure is not None:
            reconfigure(encoding="utf-8", errors="strict", newline="\n")


def _meta(name: str, row: dict) -> Block:
    return Block("meta", name, list(row), [row])


def _emit(blocks: list[Block]) -> None:
    sys.stdout.write(dumps_canonical(Document(blocks=blocks)) + "\n")


def _joined(items) -> str | None:
    items = list(items)
    return ",".join(items) if items else None


def _kv(m: Mapping) -> str | None:
    return _joined(f"{k}={v}" for k, v in m.items())


def main(argv: list[str]) -> int:
    _utf8_streams()
    ap = _Parser(prog="python -m isongraph_ext.igdm")
    sub = ap.add_subparsers(dest="cmd", required=True)
    v = sub.add_parser("verify")
    v.add_argument("graph", type=Path)
    v.add_argument("binding", type=Path)
    v.add_argument("payload", type=Path)
    v.add_argument("--allow-retired", action="store_true")
    a = ap.parse_args(argv)
    try:
        graph, binding_bytes, payload_bytes = a.graph.read_bytes(), a.binding.read_bytes(), a.payload.read_bytes()
    except OSError as e:
        ap.print_usage(sys.stderr)
        sys.stderr.write(f"error: {e}\n")
        return EX_USAGE
    try:
        binding = ExtensionBinding.from_ison(binding_bytes)
        result, dm, decision = verify_and_admit(payload_bytes, binding, graph, policy=AdmissionPolicy(allow_retired_baselines=a.allow_retired))
    except (IGXError, IGDMError) as e:
        _emit([_meta("error", {"code": e.code, "field": e.field, "detail": e.detail, "step": e.step})])
        return 1
    _emit([
        _meta("verification", {
            "binding_digest": result.binding_digest, "semantic_verification": result.semantic_verification,
            "source_canonical": result.source_canonical, "source_semantic_eligibility": result.source_semantic_eligibility,
            "source_semantic_reason": result.source_semantic_reason, "steps_passed": _joined(result.steps_passed),
            "unrecognized_optional": _joined(c.extension for c in result.unrecognized_optional)}),

                            "receipt_posture": decision.receipt_posture, "reasons": _joined(decision.reasons)}),
    ])
    return 0 if decision.admitted else 2

    sys.exit(main(sys.argv[1:]))

def _details(detail):
    d = {}
    for item in (detail or "").split(";"):
        if ":" in item:
            k, v = item.split(":", 1)
            d.setdefault(k.strip(), []).append(v.strip())
    return d


def check(id_, target_rel, binding_text, payload_text, expect, step, detail) -> None:
    d = _details(detail)
    limits = Limits(**{k: int(v) for item in d.get("limits", []) for k, v in [item.split("=")]})
    tokens = expect.split("+")
    try:
        result, dm, decision = verify_and_admit(payload_text.encode("utf-8"), ExtensionBinding.from_ison(binding_text), GRAPHS[target_rel], limits=limits, baselines=B)
    except (IGXError, IGDMError) as e:
        s = "D0" if isinstance(e, IGXError) else e.step
        assert tokens[0].startswith("reject:") and (e.code, s) == (tokens[0][len("reject:"):], step), (id_, e)
        return
    assert "verify_ok" in tokens, id_
    for csv in d.get("roles", []):
        assert dm.unrecognized_roles == tuple(csv.split(",")), (id_, dm.unrecognized_roles)
    assert dm.object_resolution == ("skipped" if "skipped" in d.get("resolution", []) else "full"), id_
    for t in tokens:
        if t == "admit_ok":
            assert decision.admitted, (id_, decision)
        elif t.startswith("admit:"):
            assert (decision.admitted, decision.code, decision.step) == (False, t[len("admit:"):], step), (id_, decision)


def write_all() -> None:
    for sub in ("graphs/synthetic", "bindings", "payloads"):
        shutil.rmtree(VEC / sub, ignore_errors=True)
        (VEC / sub).mkdir(parents=True)
    for rel, data in GRAPHS.items():
        (VEC / "graphs" / rel).write_bytes(data)
    rows = []
    for id_, target_rel, binding_text, payload_text, expect, step, detail in VECTORS:
        if binding_text is None:
            binding_text = bind_text(payload_text, GRAPHS[target_rel])
        check(id_, target_rel, binding_text, payload_text, expect, step, detail)
        slug = id_.replace("/", "--")
        (VEC / "payloads" / f"{slug}.igdm.ison").write_bytes(payload_text.encode("utf-8"))
        (VEC / "bindings" / f"{slug}.igx.ison").write_bytes(binding_text.encode("utf-8"))
        rows.append({"id": id_, "target": target_rel, "binding": f"{slug}.igx.ison", "payload": f"{slug}.igdm.ison", "expect": expect, "step": step, "detail": detail})
    doc = Document(blocks=[Block("meta", "corpus", ["datamatics_spec", "version"], [{"datamatics_spec": "IGDM/1", "version": 1}]),

    print(f"wrote {len(rows)} vectors")
