import pytest
from ison_parser import Block, Document, dumps_canonical

from isongraph_ext.strict_ison import BlockShape, StrictError, read_canonical

SHAPES = (BlockShape("meta", "m", ("a", "b"), single_row=True), BlockShape("table", "t", ("x", "y")))
BIG = 1 << 20


def good() -> str:
    return dumps_canonical(Document(blocks=[Block("meta", "m", ["a", "b"], [{"a": 1, "b": "two"}]),
                                            Block("table", "t", ["x", "y"], [{"x": 1, "y": None}, {"x": 2, "y": "q"}])]))


def read(text, **kw):
    return read_canonical(text, SHAPES, max_bytes=kw.pop("max_bytes", BIG), max_rows=kw.pop("max_rows", 100))


def refused(text, stage, kind="malformed", field=None, **kw):
    with pytest.raises(StrictError) as e:
        read(text, **kw)
    assert (e.value.stage, e.value.kind) == (stage, kind), e.value
    if field is not None:
        assert e.value.field == field, e.value
    return e.value

# isongraph-ext/src/isongraph_ext/strict_ison.py
"""Strict canonical-ISON reader shared by IGDC/1 (26 §4, C0-C1) and IGAR/1 (25 §3, R0-R1).

ison-py parses more than these contracts allow, and what it accepts is not always visible afterwards (probed against ison-py 1.2.0):
a repeated column header is kept in ``Block.fields`` while the row dict silently keeps the LAST value; a short row is padded with
``None``; ``---`` summary lines, ``name:int`` / ``name:computed`` column declarations and ``:ref`` cells parse without error, and
the canonical writer re-emits all of them unchanged. So a byte-for-byte re-emission check alone does not prove a document has the
shape a contract names. This module is that shape check, written once, with the byte budget applied BEFORE any parsing.

It knows nothing about IGDC or IGAR vocabularies; callers pass the exact blocks and columns they allow.
"""
from __future__ import annotations

import math
from collections.abc import Sequence
from dataclasses import dataclass

from ison_parser import dumps_canonical, loads

_SCALARS = (str, int, float, bool, type(None))      # exact types only: ison-py's Reference (":user:7") is none of them


class StrictError(Exception):
    """``stage`` is "bytes" (C0/R0) or "shape" (C1/R1); ``kind`` is "malformed" or "limit"."""

    def __init__(self, stage: str, kind: str, field: str, detail: str) -> None:
        super().__init__(f"{stage} {kind} {field}: {detail}")
        self.stage, self.kind, self.field, self.detail = stage, kind, field, detail


@dataclass(frozen=True)
class BlockShape:
    kind: str                       # "meta" | "table"
    name: str
    fields: tuple[str, ...]         # the exact column set; declared order is irrelevant here (the canonical writer sorts)
    single_row: bool = False        # meta blocks: exactly one row


def _bytes_fail(kind: str, field: str, detail: str) -> StrictError:
    return StrictError("bytes", kind, field, detail)


def _shape_fail(field: str, detail: str) -> StrictError:
    return StrictError("shape", "malformed", field, detail)


def read_canonical(data: bytes | str, shapes: Sequence[BlockShape], *, max_bytes: int, max_rows: int) -> tuple[str, dict[tuple[str, str], list[dict]]]:
    """Return ``(text, rows_by_block)`` for a document that is canonical AND has exactly the allowed shape, or raise StrictError.

    Order, first failure wins: byte budget (before any decoding or parsing) -> UTF-8 -> CR / BOM -> parse -> canonical re-emission
    -> block set -> column declarations -> summary / typed columns -> row budget and cardinality -> cell types.
    """
    if isinstance(data, str):
        try:
            raw = data.encode("utf-8")
        except UnicodeEncodeError as e:
            raise _bytes_fail("malformed", "document", f"text is not UTF-8-encodable: {e}") from None
    elif isinstance(data, (bytes, bytearray)):
        raw = bytes(data)
    else:
        raise TypeError(f"data must be bytes or str, got {type(data).__name__}")
    if len(raw) > max_bytes:                                                    # budget first: nothing below runs on oversize input
        raise _bytes_fail("limit", "size", f"{len(raw)} bytes > {max_bytes}")
    try:
        text = raw.decode("utf-8")
    except UnicodeDecodeError as e:
        raise _bytes_fail("malformed", "document", f"not UTF-8: {e}") from None
    if text.startswith("﻿"):
        raise _bytes_fail("malformed", "canonical_form", "byte-order mark present")
    if "\r" in text:
        raise _bytes_fail("malformed", "canonical_form", "CR bytes present; the canonical writer emits LF only")
    try:
        doc = loads(text)
        again = dumps_canonical(doc)
    except Exception as e:                                                      # any parser or writer fault is "not a document"
        raise _bytes_fail("malformed", "document", f"not ISON: {e}") from None
    if again != text:
        raise _bytes_fail("malformed", "canonical_form", "bytes are not the canonical writer's output")

    allowed = {(s.kind, s.name): s for s in shapes}
    found: dict[tuple[str, str], object] = {}
    for b in doc.blocks:
        key = (b.kind, b.name)
        if key not in allowed:
            raise _shape_fail(f"{b.kind}.{b.name}", "unexpected block")
        if key in found:
            raise _shape_fail(f"{b.kind}.{b.name}", "duplicate block")
        found[key] = b
    out: dict[tuple[str, str], list[dict]] = {}
    for key, shape in allowed.items():
        b = found.get(key)
        where = f"{key[0]}.{key[1]}"
        if b is None:
            raise _shape_fail(where, "missing block")
        declared = list(b.fields)
        for name in declared:
            if declared.count(name) > 1:
                raise _shape_fail(f"{where}.{name}", "column declared more than once")
        if set(declared) != set(shape.fields):
            missing, extra = sorted(set(shape.fields) - set(declared)), sorted(set(declared) - set(shape.fields))
            raise _shape_fail(where, f"columns differ from the contract (missing {missing}, unexpected {extra})")
        if getattr(b, "summary", None) is not None:
            raise _shape_fail(where, "summary row (---) is not part of this contract")
        for fi in b.field_info:
            if fi.type is not None or fi.is_computed:
                raise _shape_fail(f"{where}.{fi.name}", "typed or computed column declaration is not part of this contract")
        if len(b.rows) > max_rows:
            raise StrictError("shape", "limit", "max_rows_per_table", f"{where} has {len(b.rows)} rows > {max_rows}")
        if shape.single_row and len(b.rows) != 1:
            raise _shape_fail(where, "exactly one row required")
        for row in b.rows:
            if set(row) != set(shape.fields):
                raise _shape_fail(where, "a row does not carry exactly the declared columns")
            for col, v in row.items():
                if type(v) not in _SCALARS:
                    raise _shape_fail(f"{where}.{col}", f"cell is not a scalar ({type(v).__name__})")
                if isinstance(v, float) and not math.isfinite(v):
                    raise _shape_fail(f"{where}.{col}", "non-finite number")
        out[key] = [dict(r) for r in b.rows]
    return text, out
```

"""Typed-address mapping into igs-addr/1 (26 §1). Reads the TYPE of a value, never its spelling."""
from __future__ import annotations

import re
from dataclasses import dataclass

from ..igpv.records import INT64_MAX, INT64_MIN, EdgeAddress, NodeRef, is_identifier, is_int, is_str
from ..igpv.resolve import canonical_endpoints, node_ref_key
from .errors import IGDCError

_INT64_DECIMAL = re.compile(r"0|-?[1-9][0-9]*")      # ASCII digits only; no -0, +5, leading zeros
_ORDINAL = re.compile(r"0|[1-9][0-9]*")


def _bad(field: str, detail: str) -> IGDCError:
    return IGDCError("INVALID_ADDRESS", field, detail, "C2")


def _utf8_ok(s: str) -> bool:
    try:
        s.encode("utf-8")
        return True
    except UnicodeEncodeError:
        return False


def tag_id(v: int | str) -> str:
    if is_int(v):
        if not INT64_MIN <= v <= INT64_MAX:
            raise _bad("node_id", f"integer {v} lies outside INT64")
        return f"i:{v}"
    if is_str(v):
        if not _utf8_ok(v):
            raise _bad("node_id", "string is not UTF-8-encodable (lone surrogate)")
        return f"s:{v}"
    raise _bad("node_id", "id must be an INT64 integer or a string (booleans, floats, null, bytes are not ids)")


def parse_id(tag: str) -> int | str:
    if not is_str(tag) or tag[:2] not in ("i:", "s:"):
        raise _bad("node_id", "a tagged id starts with 'i:' or 's:'; a bare value is not an identity")
    body = tag[2:]
    if tag[0] == "s":
        if not _utf8_ok(body):
            raise _bad("node_id", "string is not UTF-8-encodable (lone surrogate)")
        return body
    if _INT64_DECIMAL.fullmatch(body) is None:
        raise _bad("node_id", f"{body!r} is not a canonical INT64 decimal")
    n = int(body)
    if not INT64_MIN <= n <= INT64_MAX:
        raise _bad("node_id", f"integer {n} lies outside INT64")
    return n


def semantic_key(node_type: str, tagged_id: str) -> tuple:
    return node_ref_key(NodeRef(node_type, parse_id(tagged_id)))


@dataclass(frozen=True)
class TypedNode:
    node_type: str
    node_id: str


@dataclass(frozen=True)
class TypedEdge:
    source: TypedNode
    relation: str
    target: TypedNode
    ordinal: str


def node_address(ref: NodeRef) -> TypedNode:
    if not is_identifier(ref.node_type):
        raise _bad("node_type", 'must be a Core identifier (non-empty, no whitespace, none of : | " \\)')
    return TypedNode(ref.node_type, tag_id(ref.node_id))


def edge_address(e: EdgeAddress, *, directed: bool, multigraph: bool) -> TypedEdge:
    if not is_identifier(e.relation):
        raise _bad("relation", 'must be a Core identifier (non-empty, no whitespace, none of : | " \\)')
    if not (is_str(e.ordinal) and _ORDINAL.fullmatch(e.ordinal)):
        raise _bad("ordinal", "canonical decimal string 0|[1-9][0-9]*")
    if not multigraph and e.ordinal != "0":
        raise _bad("ordinal", 'a simple graph carries ordinal "0"')
    src, tgt = node_address(e.source), node_address(e.target)
    if not directed and canonical_endpoints(e.source, e.target) != (e.source, e.target):
        raise _bad("source_id", "endpoints are not in canonical semantic order for an undirected graph")
    return TypedEdge(src, e.relation, tgt, e.ordinal)


def edge_semantic_key(e: TypedEdge) -> tuple:
    """Edge order (26 §1.3): source, relation bytes, target, ordinal bytes ("10" sorts before "2")."""
    return (semantic_key(e.source.node_type, e.source.node_id), e.relation.encode("utf-8"),
            semantic_key(e.target.node_type, e.target.node_id), e.ordinal.encode("ascii"))


class TwinGuard:
    """ID-TWIN-INVARIANT (26 §1.4, inherited from IGDM/1 D1): within one node_type, i:<d> and s:<d> never both occur."""

    def __init__(self) -> None:
        self._ints: dict[tuple[str, str], str] = {}
        self._strs: dict[tuple[str, str], str] = {}

    def see(self, ref: NodeRef, field: str) -> None:
        if is_int(ref.node_id):
            k = (ref.node_type, str(ref.node_id))
            if k in self._strs:
                raise IGDCError("TWIN_IDS", field, f"INT64 id {ref.node_id} is a twin of the STRING id at {self._strs[k]}", "C2")
            self._ints.setdefault(k, field)
        else:
            k = (ref.node_type, ref.node_id)
            if k in self._ints:
                raise IGDCError("TWIN_IDS", field, f"STRING id {ref.node_id!r} is a twin of the INT64 id at {self._ints[k]}", "C2")
            self._strs.setdefault(k, field)

"""IGDC/1 hint document: model, canonical wire (26 §2) and consumer checks C0-C3 (26 §4).

Reading goes through `strict_ison.read_canonical` (budget, canonical form, exact blocks, unique columns, no summary or typed
columns, scalar cells). Field vocabularies are IGDM/1's own: `_vocab`, `_common_hints`, `_untok` and `check_references` are
imported from `igdm.payload` and reused, never copied, so the two contracts cannot drift. Nothing in igdm is modified.
"""
from __future__ import annotations

from dataclasses import dataclass
from types import SimpleNamespace
from typing import Any

from ison_parser import Block, Document, dumps_canonical

from ..igdm import records as R
from ..igdm.errors import IGDMError
from ..igdm.payload import DatamaticsPayload, _common_hints, _is_hex64, _is_key, _untok, _vocab, check_references
from ..igpv.records import EdgeAddress, NodeRef
from ..igx.digest import sha256_hex
from ..strict_ison import BlockShape, StrictError, read_canonical
from .address import TwinGuard, edge_address, node_address, parse_id
from .errors import IGDCError

X_SPLIT = {"nodes": ("role", "ring", "a11y_role", "shape_role"), "edges": ("role", "pattern_role"), "groups": ("role",), "layout": ("family",)}


def _expand(cols: tuple[str, ...], xs: tuple[str, ...]) -> tuple[str, ...]:
    """Insert `<c>_x` right after each x-capable column `c` (26 §2.2)."""
    return tuple(y for c in cols for y in ((c, c + "_x") if c in xs else (c,)))


META_COLUMNS = ("consumption_spec", "address_scheme", "igdm_payload_sha256", "target_artifact_sha256", "target_directed", "target_multigraph",
                "target_serialization_profile")
LAYOUT_COLUMNS = _expand(R.LAYOUT_FIELDS, X_SPLIT["layout"])
NODE_COLUMNS = ("node_type", "node_id") + _expand(R.NODE_HINTS, X_SPLIT["nodes"])
EDGE_COLUMNS = ("source_type", "source_id", "relation", "target_type", "target_id", "ordinal") + _expand(R.EDGE_HINTS, X_SPLIT["edges"])
GROUP_COLUMNS = ("group_id", "intent") + _expand(R.GROUP_FIELDS[2:], X_SPLIT["groups"])
MEMBER_COLUMNS = R.MEMBER_FIELDS

SHAPES = (BlockShape("meta", "igdc", META_COLUMNS, True), BlockShape("meta", "layout", LAYOUT_COLUMNS, True),
          BlockShape("table", "nodes", NODE_COLUMNS), BlockShape("table", "edges", EDGE_COLUMNS),
          BlockShape("table", "groups", GROUP_COLUMNS), BlockShape("table", "members", MEMBER_COLUMNS))


def _c1(field: str, detail: str) -> IGDCError:
    return IGDCError("MALFORMED_CONSUMPTION", field, detail, "C1")


def _from_strict(e: StrictError) -> IGDCError:
    step = "C0" if e.stage == "bytes" else "C1"
    if e.kind == "limit":
        return IGDCError("DATAMATICS_LIMIT_EXCEEDED", "max_payload_bytes" if e.field == "size" else e.field, e.detail, step)
    return IGDCError("MALFORMED_CONSUMPTION", e.field, e.detail, step)


def _need(cond: bool, field: str, detail: str) -> None:
    if not cond:
        raise _c1(field, detail)


def _merged(row: dict, xcols: tuple[str, ...], where: str) -> dict:
    """Fold each `<c>`/`<c>_x` pair back into one `<c>` for IGDM/1's field rules; refuse every §2.2 violation on the way."""
    out = dict(row)
    for c in xcols:
        v, x = out.pop(c), out.pop(c + "_x")
        _need(v is None or x is None, f"{where}.{c}", f"{c} and {c}_x are both set")
        if x is not None:
            _need(isinstance(x, str) and R.X_TOKEN.fullmatch(x) is not None, f"{where}.{c}_x", "an x- token (x-[a-z0-9-]+)")
            out[c] = x
        else:
            _need(not (isinstance(v, str) and R.X_TOKEN.fullmatch(v)), f"{where}.{c}", f"an x- token belongs in {c}_x, not in the closed column")
            out[c] = v
    return out


def _igdm_rules(where: str, row: dict, cols: tuple[str, ...]) -> None:
    try:
        obj = SimpleNamespace(**{c: (_untok(row[c], f"{where}.{c}") if c in R.BOOL_COLUMNS else row[c]) for c in cols})
        _common_hints(obj, where, cols)
    except IGDMError as e:
        raise _c1(e.field, e.detail) from None


@dataclass(frozen=True)
class HintSet:
    meta: dict
    layout: dict
    nodes: tuple[dict, ...] = ()
    edges: tuple[dict, ...] = ()
    groups: tuple[dict, ...] = ()
    members: tuple[dict, ...] = ()

    def to_document(self) -> Document:
        return Document(blocks=[
            Block("meta", "igdc", list(META_COLUMNS), [dict(self.meta)]), Block("meta", "layout", list(LAYOUT_COLUMNS), [dict(self.layout)]),
            Block("table", "nodes", list(NODE_COLUMNS), [dict(r) for r in self.nodes]), Block("table", "edges", list(EDGE_COLUMNS), [dict(r) for r in self.edges]),
            Block("table", "groups", list(GROUP_COLUMNS), [dict(r) for r in self.groups]), Block("table", "members", list(MEMBER_COLUMNS), [dict(r) for r in self.members])])

    def to_ison(self) -> str:
        return dumps_canonical(self.to_document())

    @property
    def digest(self) -> str:
        return sha256_hex(self.to_ison())

    @classmethod
    def from_ison(cls, data: bytes | str, *, limits: R.Limits = R.DEFAULT_LIMITS) -> "HintSet":
        """C0 and C1. The byte budget is `limits.max_payload_bytes`, applied before any parsing (26 §4)."""
        try:
            text, b = read_canonical(data, SHAPES, max_bytes=limits.max_payload_bytes, max_rows=limits.max_rows_per_table)
        except StrictError as e:
            raise _from_strict(e) from None
        h = cls(b[("meta", "igdc")][0], b[("meta", "layout")][0], tuple(b[("table", "nodes")]), tuple(b[("table", "edges")]),
                tuple(b[("table", "groups")]), tuple(b[("table", "members")]))
        h._check_fields()
        if h.to_ison() != text:                                       # the model re-emits what the writer would: a second, independent check
            raise IGDCError("MALFORMED_CONSUMPTION", "canonical_form", "document is not the canonical writer's output for this hint set", "C0")
        return h

    def _check_fields(self) -> None:
        m = self.meta
        _need(m["consumption_spec"] == "IGDC/1", "meta.igdc.consumption_spec", "must be IGDC/1")
        _need(m["address_scheme"] == "igs-addr/1", "meta.igdc.address_scheme", "must be igs-addr/1")
        for col in ("igdm_payload_sha256", "target_artifact_sha256"):
            _need(_is_hex64(m[col]), f"meta.igdc.{col}", "64 lowercase hex")
        _need(m["target_serialization_profile"] in R.PROFILES, "meta.igdc.target_serialization_profile", f"not one of {R.PROFILES}")
        for col in ("target_directed", "target_multigraph"):
            _need(isinstance(m[col], str) and m[col] in ("true", "false"), f"meta.igdc.{col}", 'the string token "true" or "false"')
        lay = _merged(self.layout, X_SPLIT["layout"], "meta.layout")
        try:
            for col in ("family", "direction", "stability"):
                _vocab(lay[col], col, "meta.layout")
        except IGDMError as e:
            raise _c1(e.field, e.detail) from None
        _need(lay["direction"] is None or lay["family"] is not None, "meta.layout.direction", "direction needs a family")
        for r in self.nodes:
            w = f"table.nodes.{r['node_type']}:{r['node_id']}"
            _need(isinstance(r["node_type"], str) and isinstance(r["node_id"], str), w, "node_type and node_id are strings")
            _igdm_rules(w, _merged(r, X_SPLIT["nodes"], w), R.NODE_HINTS)
        for r in self.edges:
            w = f"table.edges.{r['source_type']}:{r['source_id']}/{r['relation']}/{r['target_type']}:{r['target_id']}/{r['ordinal']}"
            for c in ("source_type", "source_id", "relation", "target_type", "target_id", "ordinal"):
                _need(isinstance(r[c], str), f"{w}.{c}", "a string")
            _igdm_rules(w, _merged(r, X_SPLIT["edges"], w), R.EDGE_HINTS)
        for r in self.groups:
            w = f"table.groups.{r['group_id']}"
            g = _merged(r, X_SPLIT["groups"], w)
            _need(_is_key(r["group_id"]), f"{w}.group_id", "key token")
            _need(r["parent_group_id"] is None or _is_key(r["parent_group_id"]), f"{w}.parent_group_id", "null or key token")
            try:
                _vocab(r["intent"], "intent", w)
            except IGDMError as e:
                raise _c1(e.field, e.detail) from None
            _need(r["intent"] is not None, f"{w}.intent", "required")
            _igdm_rules(w, g, ("label", "role", "collapsed", "visibility", "lod_min", "a11y_text"))
        for r in self.members:
            w = f"table.members.{r['group_id']}/{r['node_type']}:{r['node_id']}"
            _need(_is_key(r["group_id"]) and isinstance(r["node_type"], str) and isinstance(r["node_id"], str), w, "group_id key token; node_type and node_id strings")

    def check(self, *, limits: R.Limits = R.DEFAULT_LIMITS) -> None:
        """C2 (addresses) then C3 (references). Needs no target and no Python runtime beyond this package."""
        directed, multigraph = self.meta["target_directed"] == "true", self.meta["target_multigraph"] == "true"
        twins = TwinGuard()

        def parsed(field: str, node_type: str, tagged: str) -> NodeRef:
            try:
                return NodeRef(node_type, parse_id(tagged))
            except IGDCError as e:
                raise IGDCError(e.code, field, e.detail, "C2") from None

        def dup(seen: set, key: Any, field: str, what: str) -> None:
            if key in seen:
                raise IGDCError("INVALID_ADDRESS", field, f"duplicate {what}", "C2")
            seen.add(key)

        seen_n: set = set()
        for r in self.nodes:
            w = f"table.nodes.{r['node_type']}:{r['node_id']}"
            ref = parsed(f"{w}.node_id", r["node_type"], r["node_id"])
            node_address(ref)
            dup(seen_n, ref, f"{w}.node_id", "node row")
            twins.see(ref, f"{w}.node_id")
        seen_e: set = set()
        for r in self.edges:
            w = f"table.edges.{r['source_type']}:{r['source_id']}/{r['relation']}/{r['target_type']}:{r['target_id']}/{r['ordinal']}"
            e = EdgeAddress(parsed(f"{w}.source_id", r["source_type"], r["source_id"]), r["relation"],
                            parsed(f"{w}.target_id", r["target_type"], r["target_id"]), r["ordinal"])
            edge_address(e, directed=directed, multigraph=multigraph)
            dup(seen_e, e, f"{w}.relation", "edge row")
            twins.see(e.source, f"{w}.source_id")
            twins.see(e.target, f"{w}.target_id")
        seen_g: set = set()
        for r in self.groups:
            dup(seen_g, r["group_id"], f"table.groups.{r['group_id']}.group_id", "group id")
        seen_m: set = set()
        members: list[tuple[str, NodeRef]] = []
        for r in self.members:
            w = f"table.members.{r['group_id']}/{r['node_type']}:{r['node_id']}"
            ref = parsed(f"{w}.node_id", r["node_type"], r["node_id"])
            node_address(ref)
            dup(seen_m, (r["group_id"], ref), f"{w}.node_id", "membership")
            twins.see(ref, f"{w}.node_id")
            members.append((r["group_id"], ref))
        # C3: IGDM/1's own D3 over exactly the groups and members this document carries (unknown parent / member group, cycle, depth)
        p = DatamaticsPayload(self.meta["target_artifact_sha256"], self.meta["target_serialization_profile"],
                              groups=tuple(R.Group(g["group_id"], g["intent"], parent_group_id=g["parent_group_id"]) for g in self.groups),
                              members=tuple(R.Member(gid, ref) for gid, ref in members))
        try:
            check_references(p, limits)
        except IGDMError as e:
            raise IGDCError(e.code, e.field, e.detail, "C3") from None

import hashlib
from collections import defaultdict

import pytest
from conftest import ROOT
from ison_parser import loads

from isongraph_ext.igdc.address import TwinGuard, TypedEdge, TypedNode, edge_address, edge_semantic_key, node_address, parse_id, semantic_key, tag_id

ADDR = ROOT / "vectors" / "igs-addr"
TABLE_COUNTS = {"vectors": 71, "edges": 19, "order": 33, "edge_order": 11, "twins": 8}     # the claim a port makes: ALL of these rows pass


def lit(s: str):
    """Typed literal -> Python value; the TYPE is stated, never inferred from the digits."""
    kind, _, rest = s.partition(":")
    if kind == "int":
        return int(rest)
    if kind == "str":
        return rest
    if kind == "bool":
        return rest == "true"
    if kind == "float":
        return float(rest)
    if kind == "null":
        return None
    if kind == "surrogate":
        return chr(int(rest, 16))
    if kind == "tagged":
        return rest
    if kind == "tagged-lone":
        return "s:" + chr(int(rest, 16))
    raise AssertionError(f"unknown typed literal {s!r}")


def tables():
    doc = loads((ADDR / "manifest.ison").read_text(encoding="utf-8"))
    return {b.name: b.rows for b in doc.blocks if b.kind == "table"}


T = tables()


def expect_reject(fn, code, *a, **k):
    with pytest.raises(IGDCError) as e:
        fn(*a, **k)
    assert (e.value.code, e.value.step) == (code, "C2"), e.value


def test_manifest_counts_and_checksums():
    assert {name: len(rows) for name, rows in T.items()} == TABLE_COUNTS
    for line in (ADDR / "SHA256SUMS").read_text(encoding="utf-8").splitlines():
        digest, _, name = line.partition("  ")
        assert hashlib.sha256((ADDR / name).read_bytes()).hexdigest() == digest, name
    assert not [n for n in ("manifest.ison", "README.md") if not (ADDR / n).exists()]

        else:

            assert parse_id(expect) == v and type(parse_id(expect)) is type(v)           # lossless, and the type comes back
    elif kind == "parse":

            expect_reject(parse_id, expect.removeprefix("reject:"), lit(row["input"]))
        else:
            got, want = parse_id(lit(row["input"])), lit(expect)
            assert got == want and type(got) is type(want)
    elif kind in ("node_type", "relation"):
        v = lit(row["input"])
        ref = NodeRef(v, 1) if kind == "node_type" else None

        else:

def _groups(rows):
    g = defaultdict(list)
    for r in rows:
        g[r["id"]].append(r)
    return g


@pytest.mark.parametrize("vid,rows", _groups(T["order"]).items(), ids=lambda v: v if isinstance(v, str) else "")
def test_order_table(vid, rows):
    items = [(r["node_type"], tag_id(lit(r["node_id"])), int(r["position"])) for r in rows]
    by_semantic = sorted(items, key=lambda x: semantic_key(x[0], x[1]))
    assert [x[2] for x in by_semantic] == list(range(len(items))), vid                   # the stated ranks ARE the semantic order
    trap = {r["trap"] for r in rows}
    assert len(trap) == 1
    trap = trap.pop()
    if trap == "text":
        assert sorted(items, key=lambda x: (x[0], x[1])) != by_semantic, f"{vid}: file/text order must differ from semantic order"
    elif trap == "utf16":
        assert sorted(items, key=lambda x: (x[0].encode("utf-16-be"), x[1].encode("utf-16-be", "surrogatepass"))) != by_semantic
    else:
        assert trap == "none", f"unknown trap {trap!r}"

"""IGDC/1 derivation (26 §3): a pure function of a verified IGDM/1 payload and two facts about the target."""
from __future__ import annotations

from collections.abc import Mapping

from ison_graph.multigraph import MultiGraph

from ..igdm import records as R
from ..igdm.payload import DatamaticsPayload
from ..igdm.verify import verify as igdm_verify
from ..igpv.resolve import load_graph
from ..igx.baselines import BASELINES, Baseline
from ..igx.binding import ExtensionBinding
from ..igx.digest import sha256_hex
from ..igx.registry import DEFAULT_REGISTRY, CapabilityRegistry

from .errors import IGDCError
from .hints import EDGE_COLUMNS, GROUP_COLUMNS, LAYOUT_COLUMNS, NODE_COLUMNS, X_SPLIT, HintSet


def _tok(v: bool | None) -> str | None:
    return None if v is None else ("true" if v else "false")


def _split(value, col: str, out: dict, xcols: tuple[str, ...]) -> None:
    """26 §2.2: a closed value stays in `col`; an x- token moves to `col_x`; never both."""
    if col in xcols and isinstance(value, str) and R.X_TOKEN.fullmatch(value):
        out[col], out[col + "_x"] = None, value
    else:
        out[col] = value
        if col in xcols:
            out[col + "_x"] = None


def _row(cols: tuple[str, ...], values: dict, xcols: tuple[str, ...], bools: set[str]) -> dict:
    out: dict = {}
    for c in cols:
        if c.endswith("_x") and c[:-2] in xcols:
            continue                          # filled by _split
        v = values.get(c)
        if c in bools:
            v = _tok(v)
        _split(v, c, out, xcols)
    return {c: out.get(c) for c in cols}


def derive(p: DatamaticsPayload, *, directed: bool, multigraph: bool, payload_bytes: bytes) -> HintSet:
    nodes, edges, groups, members = [], [], [], []
    for n in p.nodes:
        a = node_address(n.node)
        vals = {c: getattr(n, c) for c in R.NODE_HINTS} | {"node_type": a.node_type, "node_id": a.node_id}
        nodes.append(_row(NODE_COLUMNS, vals, X_SPLIT["nodes"], R.BOOL_COLUMNS))
    for e in p.edges:
        a = edge_address(e.edge, directed=directed, multigraph=multigraph)
        vals = {c: getattr(e, c) for c in R.EDGE_HINTS} | {"source_type": a.source.node_type, "source_id": a.source.node_id, "relation": a.relation,
                                                            "target_type": a.target.node_type, "target_id": a.target.node_id, "ordinal": a.ordinal}
        edges.append(_row(EDGE_COLUMNS, vals, X_SPLIT["edges"], R.BOOL_COLUMNS))
    for g in p.groups:
        groups.append(_row(GROUP_COLUMNS, {c: getattr(g, c) for c in R.GROUP_FIELDS}, X_SPLIT["groups"], R.BOOL_COLUMNS))
    for m in p.members:
        a = node_address(m.node)
        members.append({"group_id": m.group_id, "node_type": a.node_type, "node_id": a.node_id})
    layout = _row(LAYOUT_COLUMNS, {"family": p.layout.family, "direction": p.layout.direction, "stability": p.layout.stability}, X_SPLIT["layout"], set())
    meta = {"consumption_spec": "IGDC/1", "address_scheme": "igs-addr/1", "igdm_payload_sha256": sha256_hex(payload_bytes),
            "target_artifact_sha256": p.target_artifact_sha256, "target_directed": _tok(directed), "target_multigraph": _tok(multigraph),
            "target_serialization_profile": p.target_serialization_profile}
    return HintSet(meta, layout, tuple(nodes), tuple(edges), tuple(groups), tuple(members))


def consume(payload: bytes | str, binding: ExtensionBinding, target: bytes | str, *, limits: R.Limits = R.DEFAULT_LIMITS,
            baselines: Mapping[str, Baseline] = BASELINES, registry: CapabilityRegistry = DEFAULT_REGISTRY) -> HintSet:
    payload_bytes, target_bytes = as_bytes(payload), as_bytes(target)
    _, dm = igdm_verify(payload_bytes, binding, target_bytes, limits=limits, baselines=baselines, registry=registry)   # errors propagate unwrapped
    if dm.object_resolution != "full":                                                                                  # P0
        raise IGDCError("RESOLUTION_REQUIRED", "object_resolution", "IGDC/1 needs a full verification: this one skipped object resolution (limited runtime)", "P0")
    p = DatamaticsPayload.from_ison(payload_bytes, limits=limits)
    graph = load_graph(target_bytes, binding.serialization_profile, "target")
    h = derive(p, directed=bool(getattr(graph, "directed", True)), multigraph=isinstance(graph, MultiGraph), payload_bytes=payload_bytes)
    size = len(h.to_ison().encode("utf-8"))                                                                              # P1
    if size > limits.max_payload_bytes:
        raise IGDCError("DATAMATICS_LIMIT_EXCEEDED", "max_payload_bytes", f"the derived hint document is {size} bytes > {limits.max_payload_bytes}", "P1")
    return h
```

def _tag(v):                                  # the cell type of the PARSED payload decides, never the spelling
    assert not isinstance(v, bool)
    return f"i:{v}" if isinstance(v, int) else f"s:{v}"


def _split(row, cols):
    out = {}
    for c in cols:
        v = row[c]
        isx = c in X_COLS and isinstance(v, str) and v.startswith("x-")
        out[c] = None if isx else v
        if c in X_COLS:
            out[c + "_x"] = v if isx else None
    return out


def _xcols(cols):
    return [y for c in cols for y in ([c, c + "_x"] if c in X_COLS else [c])]


def reference_derive(payload_bytes: bytes, directed: bool, multigraph: bool) -> str:
    doc = loads(payload_bytes.decode("utf-8"))
    B = {(b.kind, b.name): b for b in doc.blocks}
    m, lay = B[("meta", "igdm")].rows[0], B[("meta", "layout")].rows[0]
    tok = lambda b: "true" if b else "false"  # noqa: E731
    nodes = [{"node_type": r["node_type"], "node_id": _tag(r["node_id"]), **_split(r, NODE_H)} for r in B[("table", "nodes")].rows]
    edges = [{"source_type": r["source_type"], "source_id": _tag(r["source_id"]), "relation": r["relation"], "target_type": r["target_type"],
              "target_id": _tag(r["target_id"]), "ordinal": r["ordinal"], **_split(r, EDGE_H)} for r in B[("table", "edges")].rows]
    groups = [{"group_id": r["group_id"], "intent": r["intent"], **_split(r, GROUP_H)} for r in B[("table", "groups")].rows]
    members = [{"group_id": r["group_id"], "node_type": r["node_type"], "node_id": _tag(r["node_id"])} for r in B[("table", "members")].rows]
    f = lay["family"]
    isx = isinstance(f, str) and f.startswith("x-")
    lay2 = {"family": None if isx else f, "family_x": f if isx else None, "direction": lay["direction"], "stability": lay["stability"]}
    sha = hashlib.sha256(payload_bytes).hexdigest()
    return dumps_canonical(Document(blocks=[
        Block("meta", "igdc", META_C, [{"consumption_spec": "IGDC/1", "address_scheme": "igs-addr/1", "igdm_payload_sha256": sha,
              "target_artifact_sha256": m["target_artifact_sha256"], "target_directed": tok(directed), "target_multigraph": tok(multigraph),
              "target_serialization_profile": m["target_serialization_profile"]}]),
        Block("meta", "layout", ["family", "family_x", "direction", "stability"], [lay2]),
        Block("table", "nodes", ["node_type", "node_id"] + _xcols(NODE_H), nodes),
        Block("table", "edges", ["source_type", "source_id", "relation", "target_type", "target_id", "ordinal"] + _xcols(EDGE_H), edges),
        Block("table", "groups", ["group_id", "intent"] + _xcols(GROUP_H), groups),
        Block("table", "members", ["group_id", "node_type", "node_id"], members)]))

        hints=put("hints", "ok--igdc-wellformed-not-verified.igdc.ison", edit(lambda d: blk(d, "nodes").rows[0].update(label="changed")).encode()),
        expect="ok", detail="C0-C3 pass; only C4 or a trusted receipt ties a document to a payload")

    # (e) C4: well-formed documents that are not the derivation of the supplied inputs
    def c4(id_, fn, detail=None):
        row(id_, "derivation", hints=put("hints", f"{slug(id_)}.igdc.ison", edit(fn).encode()), expect="reject", code="DERIVATION_MISMATCH", step="C4",
            detail=detail, **inputs)

    c4("reject/igdc-derivation-mismatch", lambda d: blk(d, "nodes").rows[0].update(label="Order #999"))
    c4("reject/igdc-payload-digest-mismatch", set_first("igdc", "igdm_payload_sha256", "0" * 63 + "f"))
    c4("reject/igdc-directed-fact-wrong", set_first("igdc", "target_directed", "false"), "the target is directed; the document claims otherwise")
    c4("reject/igdc-multigraph-fact-wrong", set_first("igdc", "target_multigraph", "true"), "the target is a simple graph")
    c4("reject/igdc-row-dropped", lambda d: blk(d, "nodes").rows.pop())
    c4("reject/igdc-row-added", with_node(node_type="order", node_id="i:99", role="info"), "a well-formed row for a node the payload does not mention")

    # (f) manifest, README, SHA256SUMS
    cols = ["id", "op", "target", "binding", "payload", "hints", "expect", "code", "step", "detail"]
    for r in rows:
        assert set(r) == set(cols)
    ids = [r["id"] for r in rows]
    assert len(ids) == len(set(ids)), "duplicate vector id"
    text = dumps_canonical(Document(blocks=[Block("meta", "corpus", ["consumption_spec", "version"], [{"consumption_spec": "IGDC/1", "version": 1}]),
                                            Block("table", "vectors", cols, rows)]))
    (out / "manifest.ison").write_bytes(text.encode("utf-8"))
    (out / "README.md").write_bytes(CONSUMPTION_README.encode("utf-8"))
    files = sorted(str(p.relative_to(out)).replace("\\", "/") for p in out.rglob("*") if p.is_file() and p.name != "SHA256SUMS")
    (out / "SHA256SUMS").write_bytes("".join(f"{sha((out / n).read_bytes())}  {n}\n" for n in files).encode("utf-8"))
    counts: dict[str, int] = {}
    for r in rows:
        k = f"{r['op']}:{r['expect']}"
        counts[k] = counts.get(k, 0) + 1
    return counts
