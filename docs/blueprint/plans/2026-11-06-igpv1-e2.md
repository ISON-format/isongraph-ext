**Goal:** Implement IGPV/1 — the companion `<graph>.igpv.ison` provenance payload bound through IGX/1 as capability `igpv` — with verification P0–P5, bounded lineage queries, a CLI, and conformance vectors, in `ISON-format/isongraph-ext`.

**Tech Stack:** Python 3.12, `ison-py==1.2.0` (`ison_parser`), `ison-graph==1.4.1` (`ison_graph`, `ison_graph.multigraph`), pytest, `python -m build`, GitHub Actions (existing `ci.yml`).

## Review Focus

Input classes the spec implies that no task's tests would otherwise exercise; each is pinned to a task below.

1. A payload whose `node_id` is an ISON **float that prints like an int** (`1.0`) or a **boolean** — must be `MALFORMED_PROVENANCE`, never coerced to `1` (Task 4 `test_float_and_bool_ids_rejected`).
2. A `sources` mapping keyed by an **uppercase** or **non-64-hex** string — must be ignored as "not supplied" (posture `UNVERIFIED`), never matched case-insensitively or crash (Task 7 `test_sources_keys_must_be_lowercase_hex`).
3. An undirected target where the payload lists the edge with endpoints **reversed** — `UNRESOLVED_OBJECT` with detail containing `endpoints not canonical`, even though Core's `has_edge` would find it (Task 6 `test_undirected_reversed_endpoints_unresolved`).
4. A `lines/1` locator over bytes that **end with CRLF** — the CR belongs to the line, the final LF does not add a line (Task 3 `test_line_count_crlf_and_trailing_lf`).
5. Two payloads for **different** targets where one names the other's target as an origin graph with a **different declared size** than the real bytes — P4 must fail `SOURCE_ARTIFACT_MISMATCH` on size even though the digest matches (Task 7 `test_origin_size_mismatch`).

```python
from __future__ import annotations


class IGPVError(Exception):
    def __init__(self, code: str, field: str, detail: str, step: str) -> None:
        super().__init__(f"{step} {code} {field}: {detail}")
        self.code = code
        self.field = field
        self.detail = detail
        self.step = step
```

```python
from __future__ import annotations

import re
from dataclasses import dataclass
from typing import Any

PROVENANCE_SPEC = "IGPV/1"
CAPABILITY = "igpv"
CAPABILITY_VERSION = "1.0.0"
CAPABILITY_RANGE = "^1"
PROFILES = ("ison-v1", "isonl-v1")
ARTIFACT_KINDS = ("document", "isongraph")
PRODUCER_KINDS = ("extractor", "transformation")
OBJECT_KINDS = ("graph", "node", "edge")
DERIVATION_KINDS = ("identity", "collapse", "rewrite")
LOCATOR_SPAN_TYPES = ("bytes/1", "lines/1")

KEY_TOKEN = re.compile(r"[A-Za-z0-9][A-Za-z0-9._-]{0,63}")
HEX64 = re.compile(r"[0-9a-f]{64}")
ORDINAL = re.compile(r"0|[1-9][0-9]*")
MEDIA_TYPE = re.compile(r"[A-Za-z0-9!#$&^_.+-]+/[A-Za-z0-9!#$&^_.+-]+")
X_LOCATOR = re.compile(r"x-[a-z0-9-]+/[1-9][0-9]*")
INT64_MIN = -(2**63)
INT64_MAX = 2**63 - 1
_IDENTIFIER_FORBIDDEN = set(':|"\\')


def is_int(v: Any) -> bool:
    return isinstance(v, int) and not isinstance(v, bool)


def is_str(v: Any) -> bool:
    return isinstance(v, str)


def is_identifier(s: Any) -> bool:
    """Core's identifier rule: non-empty, no whitespace, none of : | " \\ (ison_graph._check_identifier)."""
    return is_str(s) and s != "" and not any(c.isspace() or c in _IDENTIFIER_FORBIDDEN for c in s)
```

```python
# isongraph-ext/src/isongraph_ext/igpv/locators.py
"""Source locators (18 §2): shape rule (P1) and verification posture (P5). Posture is derived, never stored."""

from __future__ import annotations

from . import records as R
from .errors import IGPVError


def line_count(data: bytes) -> int:
    """LF-delimited lines; a CR before the LF belongs to the line; a final line needs no LF; b'' has none."""
    n = data.count(b"\n")
    if data and not data.endswith(b"\n"):
        n += 1
    return n


def check_locator_shape(loc: R.Locator | None, size: int, where: str) -> None:
    def bad(col: str, detail: str) -> None:
        raise IGPVError("MALFORMED_PROVENANCE", f"{where}.{col}", detail, "P1")

    if loc is None:
        return
    t = loc.type
    if not R.is_str(t):
        bad("locator_type", "null, bytes/1, lines/1 or x-<name>/<n>")
    if t in R.LOCATOR_SPAN_TYPES:
        if loc.text is not None:
            bad("locator_text", f"must be null for {t}")
        if not (R.is_int(loc.start) and R.is_int(loc.end)):
            bad("locator_start", f"{t} needs integer start and end")
        lo = 0 if t == "bytes/1" else 1
        if not (lo <= loc.start <= loc.end <= size):
            bad("locator_end", f"{t} needs {lo} <= start <= end <= size ({size}); got [{loc.start}, {loc.end}]")
    elif R.X_LOCATOR.fullmatch(t):
        if loc.start is not None or loc.end is not None:
            bad("locator_start", "must be null for x-* locators")
        if not (R.is_str(loc.text) and loc.text != ""):
            bad("locator_text", "non-empty string required for x-* locators")
    else:
        bad("locator_type", "null, bytes/1, lines/1 or x-<name>/<n>")


def locator_posture(loc: R.Locator, data: bytes | None, where: str) -> str:
    """P5. `data` is the artifact's bytes when supplied (already digest/size-checked by the caller), else None."""
    if data is None or loc.type not in R.LOCATOR_SPAN_TYPES:
        return "UNVERIFIED"
    limit = len(data) if loc.type == "bytes/1" else line_count(data)
    if loc.end > limit:
        raise IGPVError("LOCATOR_INVALID", f"{where}.locator_end", f"{loc.type} end {loc.end} > {limit}", "P5")
    return "VERIFIED"
```

```python
# isongraph-ext/src/isongraph_ext/igpv/payload.py
"""IGPV/1 payload: wire model (18 §1), P1 field rules (check). Task 5 adds relation rules and P3 reference rules."""

from __future__ import annotations

from dataclasses import dataclass
from typing import Any

from ison_parser import Block, Document, dumps_canonical, loads

from ..igx.digest import sha256_hex
from . import records as R
from .errors import IGPVError
from .locators import check_locator_shape

META_FIELDS = ("provenance_spec", "target_artifact_sha256", "target_semantic_digest", "target_semantic_digest_algorithm",
               "target_serialization_profile")
ARTIFACT_FIELDS = ("artifact_id", "artifact_sha256", "kind", "media_type", "revision", "serialization_profile", "size", "uri")
PRODUCER_FIELDS = ("producer_id", "kind", "implementation", "version", "spec", "policy_sha256")
OBJECT_FIELDS = ("object_id", "graph", "kind", "node_type", "node_id", "source_type", "source_id", "relation", "target_type",
                 "target_id", "ordinal")
ATTRIBUTION_FIELDS = ("attribution_id", "object_id", "artifact_id", "producer_id", "locator_type", "locator_start",
                      "locator_end", "locator_text")
DERIVATION_FIELDS = ("derivation_id", "derived_object_id", "origin_object_id", "producer_id", "kind")
TABLES = (("artifacts", ARTIFACT_FIELDS, "artifact_id"), ("producers", PRODUCER_FIELDS, "producer_id"),
          ("objects", OBJECT_FIELDS, "object_id"), ("attributions", ATTRIBUTION_FIELDS, "attribution_id"),
          ("derivations", DERIVATION_FIELDS, "derivation_id"))
_NODE_COLS = ("node_type", "node_id")
_EDGE_COLS = ("source_type", "source_id", "relation", "target_type", "target_id", "ordinal")


def _err(field: str, detail: str) -> IGPVError:
    return IGPVError("MALFORMED_PROVENANCE", field, detail, "P1")


def _limit(name: str, detail: str) -> IGPVError:
    return IGPVError("PROVENANCE_LIMIT_EXCEEDED", name, detail, "P1")


def _key_bytes(value: Any) -> bytes:
    return str(value).encode("utf-8", "surrogatepass")


# ---- records -> rows ----------------------------------------------------------------------------------------

def _artifact_row(a: R.Artifact) -> dict:
    return {"artifact_id": a.artifact_id, "artifact_sha256": a.artifact_sha256, "kind": a.kind, "media_type": a.media_type,
            "revision": a.revision, "serialization_profile": a.serialization_profile, "size": a.size, "uri": a.uri}


def _producer_row(p: R.Producer) -> dict:
    return {"producer_id": p.producer_id, "kind": p.kind, "implementation": p.implementation, "version": p.version,
            "spec": p.spec, "policy_sha256": p.policy_sha256}


def _object_row(o: R.GraphObject) -> dict:
    row: dict[str, Any] = {k: None for k in OBJECT_FIELDS}
    row.update({"object_id": o.object_id, "graph": o.graph, "kind": o.kind})
    if o.node is not None:
        row.update({"node_type": o.node.node_type, "node_id": o.node.node_id})
    if o.edge is not None:
        e = o.edge
        row.update({"source_type": e.source.node_type, "source_id": e.source.node_id, "relation": e.relation,
                    "target_type": e.target.node_type, "target_id": e.target.node_id, "ordinal": e.ordinal})
    return row


def _attribution_row(a: R.Attribution) -> dict:
    loc = a.locator
    return {"attribution_id": a.attribution_id, "object_id": a.object_id, "artifact_id": a.artifact_id,
            "producer_id": a.producer_id, "locator_type": None if loc is None else loc.type,
            "locator_start": None if loc is None else loc.start, "locator_end": None if loc is None else loc.end,
            "locator_text": None if loc is None else loc.text}


def _derivation_row(d: R.Derivation) -> dict:
    return {"derivation_id": d.derivation_id, "derived_object_id": d.derived_object_id, "origin_object_id": d.origin_object_id,
            "producer_id": d.producer_id, "kind": d.kind}


# ---- rows -> records ----------------------------------------------------------------------------------------

def _object_from_row(r: dict) -> R.GraphObject:
    where = f"table.objects.{r['object_id']}"
    kind = r["kind"]
    want = {"node": set(_NODE_COLS), "edge": set(_EDGE_COLS)}.get(kind, set())
    for col in _NODE_COLS + _EDGE_COLS:
        if col not in want and r[col] is not None:
            raise _err(f"{where}.{col}", f"must be null for kind {kind!r}")
    node = R.NodeRef(r["node_type"], r["node_id"]) if kind == "node" else None
    edge = (R.EdgeAddress(R.NodeRef(r["source_type"], r["source_id"]), r["relation"],
                          R.NodeRef(r["target_type"], r["target_id"]), r["ordinal"]) if kind == "edge" else None)
    return R.GraphObject(r["object_id"], kind, r["graph"], node, edge)


def _attribution_from_row(r: dict) -> R.Attribution:
    if r["locator_type"] is None:
        for col in ("locator_start", "locator_end", "locator_text"):
            if r[col] is not None:
                raise _err(f"table.attributions.{r['attribution_id']}.{col}", "must be null when locator_type is null")
        loc = None
    else:
        loc = R.Locator(r["locator_type"], r["locator_start"], r["locator_end"], r["locator_text"])
    return R.Attribution(r["attribution_id"], r["object_id"], r["artifact_id"], r["producer_id"], loc)


@dataclass(frozen=True)
class ProvenancePayload:
    target_artifact_sha256: str
    target_serialization_profile: str
    artifacts: tuple[R.Artifact, ...] = ()
    producers: tuple[R.Producer, ...] = ()
    objects: tuple[R.GraphObject, ...] = ()
    attributions: tuple[R.Attribution, ...] = ()
    derivations: tuple[R.Derivation, ...] = ()

    def to_document(self) -> Document:
        def table(name: str, fields: tuple[str, ...], key: str, rows: list[dict]) -> Block:
            return Block(kind="table", name=name, fields=list(fields), rows=sorted(rows, key=lambda r: _key_bytes(r[key])))

        meta = {"provenance_spec": R.PROVENANCE_SPEC, "target_artifact_sha256": self.target_artifact_sha256,
                "target_semantic_digest": None, "target_semantic_digest_algorithm": None,
                "target_serialization_profile": self.target_serialization_profile}
        return Document(blocks=[
            Block(kind="meta", name="igpv", fields=list(META_FIELDS), rows=[meta]),
            table("artifacts", ARTIFACT_FIELDS, "artifact_id", [_artifact_row(a) for a in self.artifacts]),
            table("producers", PRODUCER_FIELDS, "producer_id", [_producer_row(p) for p in self.producers]),
            table("objects", OBJECT_FIELDS, "object_id", [_object_row(o) for o in self.objects]),
            table("attributions", ATTRIBUTION_FIELDS, "attribution_id", [_attribution_row(a) for a in self.attributions]),
            table("derivations", DERIVATION_FIELDS, "derivation_id", [_derivation_row(d) for d in self.derivations]),
        ])

    def to_ison(self) -> str:
        self.check()
        return dumps_canonical(self.to_document())

    @property
    def digest(self) -> str:
        return sha256_hex(self.to_ison())

    @classmethod
    def from_ison(cls, text: str | bytes, *, limits: R.Limits = R.DEFAULT_LIMITS) -> "ProvenancePayload":
        if isinstance(text, bytes):
            data = text
            try:
                text = data.decode("utf-8")
            except UnicodeDecodeError as e:
                raise _err("document", f"not UTF-8: {e}") from None
        else:
            try:
                data = text.encode("utf-8")
            except UnicodeEncodeError as e:
                raise _err("document", f"text is not UTF-8-encodable: {e}") from None
        if len(data) > limits.max_payload_bytes:
            raise _limit("max_payload_bytes", f"{len(data)} > {limits.max_payload_bytes}")
        if "\r" in text:
            raise _err("canonical_form", "CR bytes present; the canonical writer emits LF only")
        try:
            doc = loads(text)
        except Exception as e:
            raise _err("document", f"not ISON: {e}") from None
        blocks = _collect_blocks(doc)
        meta = _one_row(blocks[("meta", "igpv")], "meta.igpv", META_FIELDS)
        if meta["provenance_spec"] != R.PROVENANCE_SPEC:
            raise _err("meta.igpv.provenance_spec", f"must be {R.PROVENANCE_SPEC}")
        if meta["target_semantic_digest"] is not None or meta["target_semantic_digest_algorithm"] is not None:
            raise _err("meta.igpv.target_semantic_digest", "reserved: must be null until Semantic/2 freezes")
        rows: dict[str, list[dict]] = {}
        for name, fields, _key in TABLES:
            b = blocks[("table", name)]
            if set(b.fields) != set(fields):
                raise _err(f"table.{name}", f"fields {sorted(b.fields)} != {sorted(fields)}")
            if len(b.rows) > limits.max_rows_per_table:
                raise _limit("max_rows_per_table", f"table.{name} has {len(b.rows)} rows > {limits.max_rows_per_table}")
            rows[name] = b.rows
        payload = cls(
            target_artifact_sha256=meta["target_artifact_sha256"], target_serialization_profile=meta["target_serialization_profile"],
            artifacts=tuple(R.Artifact(r["artifact_id"], r["artifact_sha256"], r["kind"], r["size"], r["media_type"], r["revision"],
                                       r["uri"], r["serialization_profile"]) for r in rows["artifacts"]),
            producers=tuple(R.Producer(r["producer_id"], r["kind"], r["implementation"], r["version"], r["spec"], r["policy_sha256"])
                            for r in rows["producers"]),
            objects=tuple(_object_from_row(r) for r in rows["objects"]),
            attributions=tuple(_attribution_from_row(r) for r in rows["attributions"]),
            derivations=tuple(R.Derivation(r["derivation_id"], r["derived_object_id"], r["origin_object_id"], r["producer_id"], r["kind"])
                              for r in rows["derivations"]),
        )
        if payload.to_ison() != text:      # to_ison() runs check() first
            raise _err("canonical_form", "payload bytes are not the canonical writer's output for this document (row order included)")
        return payload

    def check(self) -> None:
        """P1 rules that need no bytes (18 §4.1 P1): field rules, then relation rules."""
        _check_fields(self)
        _check_relations(self)


def _collect_blocks(doc: Document) -> dict[tuple[str, str], Block]:
    required = {("meta", "igpv")} | {("table", name) for name, _f, _k in TABLES}
    out: dict[tuple[str, str], Block] = {}
    for b in doc.blocks:
        key = (b.kind, b.name)
        if key not in required:
            raise _err(f"{b.kind}.{b.name}", "unexpected block")
        if key in out:
            raise _err(f"{b.kind}.{b.name}", "duplicate block")
        out[key] = b
    for key in sorted(required - set(out)):
        raise _err(f"{key[0]}.{key[1]}", "missing block")
    return out


def _one_row(b: Block, name: str, expected: tuple[str, ...]) -> dict[str, Any]:
    if len(b.rows) != 1:
        raise _err(name, f"expected exactly one row, got {len(b.rows)}")
    row = b.rows[0]
    for f in expected:
        if f not in row:
            raise _err(f"{name}.{f}", "missing field")
    for f in row:
        if f not in expected:
            raise _err(f"{name}.{f}", "unexpected field")
    return row


def _need(cond: bool, field: str, detail: str) -> None:
    if not cond:
        raise _err(field, detail)


def _check_node_ref(ref: Any, where: str, type_col: str, id_col: str) -> None:
    _need(isinstance(ref, R.NodeRef), where, "NodeRef required")
    _need(R.is_identifier(ref.node_type), f"{where}.{type_col}", 'Core identifier (non-empty, no whitespace, none of : | " \\)')
    v = ref.node_id
    if R.is_int(v):
        _need(R.INT64_MIN <= v <= R.INT64_MAX, f"{where}.{id_col}", "integer id must lie within INT64")
    else:

def _opt_str(v: Any) -> bool:
    return v is None or (R.is_str(v) and v != "")


def _is_key(v: Any) -> bool:
    return R.is_str(v) and R.KEY_TOKEN.fullmatch(v) is not None


def _is_hex64(v: Any) -> bool:
    return R.is_str(v) and R.HEX64.fullmatch(v) is not None
```

```python
# isongraph-ext/src/isongraph_ext/igpv/resolve.py
"""P4 primitives (18 §1.2, §4.1): NodeRef ordering, canonical undirected endpoints, Core loading and object lookup."""

from __future__ import annotations

from ison_graph.multigraph import MultiGraph

from ..igx.source import _load
from . import records as R
from .errors import IGPVError


def node_ref_key(ref: R.NodeRef) -> tuple:
    """13 §2.3: node_type bytes, then INT64 before STRING, ints numerically, strings by UTF-8 bytes."""
    if R.is_int(ref.node_id):
        return (ref.node_type.encode("utf-8"), 0, ref.node_id)
    return (ref.node_type.encode("utf-8"), 1, ref.node_id.encode("utf-8", "surrogatepass"))


def canonical_endpoints(a: R.NodeRef, b: R.NodeRef) -> tuple[R.NodeRef, R.NodeRef]:
    return (a, b) if node_ref_key(a) <= node_ref_key(b) else (b, a)


def load_graph(data: bytes, profile: str, where: str):
    try:
        graph, _doc = _load(data, profile)
    except Exception as e:  # Core's own hierarchy plus decode errors: whatever it is, the objects cannot be resolved
        raise IGPVError("UNRESOLVED_OBJECT", where, f"graph does not load under the pinned Core: {e}", "P4") from None
    return graph


def resolve_object(graph, obj: R.GraphObject, where: str) -> None:
    def fail(col: str, detail: str) -> None:
        raise IGPVError("UNRESOLVED_OBJECT", f"{where}.{col}", detail, "P4")

    if obj.kind == "graph":
        return
    if obj.kind == "node":
        n = obj.node
        try:
            found = graph.has_node(n.node_type, n.node_id)
        except Exception:
            found = False
        if not found:
            fail("node_id", f"no node {n.node_type}:{n.node_id!r} in the graph")
        return
    e = obj.edge
    if not getattr(graph, "directed", True) and canonical_endpoints(e.source, e.target) != (e.source, e.target):
        fail("source_id", "endpoints not canonical for an undirected graph (13 §2.3 order required)")
    src = (e.source.node_type, e.source.node_id)
    tgt = (e.target.node_type, e.target.node_id)
    try:
        if isinstance(graph, MultiGraph):
            found = graph.has_edge(e.relation, src, tgt, key=int(e.ordinal))
        else:
            if e.ordinal != "0":
                fail("ordinal", 'simple graph edges carry ordinal "0"')
            found = graph.has_edge(e.relation, src, tgt)
    except IGPVError:
        raise
    except Exception:
        found = False
    if not found:
        fail("relation", f"no edge {src!r} -{e.relation}-> {tgt!r} ordinal {e.ordinal} in the graph")
```

```python
    return hashlib.sha256(data).hexdigest()


def target_graph() -> bytes:
    g = ISONGraph()

    base = dict(
        target_artifact_sha256=sha(target),
```

```python
"""IGPV/1 verification (18 §4.1): preflight, then P0–P5. Runs igx.verify() itself; never accepts a caller-supplied result."""

from __future__ import annotations

from collections.abc import Mapping
from dataclasses import dataclass

from ..igx.baselines import BASELINES, Baseline
from ..igx.binding import ExtensionBinding
from ..igx.digest import sha256_hex
from ..igx.registry import DEFAULT_REGISTRY, CapabilityRegistry
from ..igx.source import as_bytes
from ..igx.verify import VerificationResult
from ..igx.verify import verify as igx_verify
from ..igx.versions import parse_range, parse_version, range_admits
from . import records as R
from .errors import IGPVError
from .locators import locator_posture
from .payload import ProvenancePayload, check_references
from .resolve import load_graph, resolve_object

STEPS = ("P0", "P1", "P2", "P3", "P4", "P5")


@dataclass(frozen=True)
class ProvenanceVerification:
    payload_sha256: str
    binding_digest: str
    target_artifact_sha256: str
    object_resolution: str                  # "full" | "skipped"
    origin_resolution: Mapping[str, str]    # origin artifact_id -> "verified" | "unverified"
    locators: Mapping[str, str]             # attribution_id -> "VERIFIED" | "UNVERIFIED" (rows with a locator)
    counts: Mapping[str, int]
    steps_passed: tuple[str, ...]


def _limit(name: str, detail: str) -> IGPVError:
    return IGPVError("PROVENANCE_LIMIT_EXCEEDED", name, detail, "P1")


def _payload_bytes(payload: str | bytes | ProvenancePayload) -> bytes:
    if isinstance(payload, ProvenancePayload):
        return payload.to_ison().encode("utf-8")
    if isinstance(payload, str):
        try:
            return payload.encode("utf-8")
        except UnicodeEncodeError as e:
            raise IGPVError("MALFORMED_PROVENANCE", "document", f"text is not UTF-8-encodable: {e}", "P1") from None
    return payload


def _preflight(target: bytes, sources: Mapping[str, bytes], limits: R.Limits) -> None:
    """Byte budgets on lengths only (18 §4.1 preflight): target and the aggregate of supplied sources, before anything is
    parsed or serialized. The per-graph bound on origin graphs is applied once P1 has identified which sources are graphs."""
    if len(target) > limits.max_graph_bytes:
        raise _limit("max_graph_bytes", f"target {len(target)} > {limits.max_graph_bytes}")
    total = sum(len(data) for data in sources.values())
    if total > limits.max_sources_bytes:
        raise _limit("max_sources_bytes", f"supplied sources total {total} > {limits.max_sources_bytes}")


def _graph_budget(p: ProvenancePayload, supplied: Mapping[str, bytes], limits: R.Limits) -> None:
    """P1 (after parsing): every supplied origin graph must fit max_graph_bytes before any Core loader sees it."""
    for a in p.artifacts:
        if a.kind == "isongraph" and a.artifact_sha256 in supplied and len(supplied[a.artifact_sha256]) > limits.max_graph_bytes:
            raise _limit("max_graph_bytes", f"origin graph {a.artifact_id} {len(supplied[a.artifact_sha256])} > {limits.max_graph_bytes}")


def _p0(binding: ExtensionBinding, payload_bytes: bytes, target: bytes, baselines, registry) -> VerificationResult:
    result = igx_verify(binding, target, baselines=baselines, registry=registry)
    row = next((c for c in binding.capabilities if c.extension == R.CAPABILITY), None)
    if row is None or row.payload_sha256 is None:
        raise IGPVError("PROVENANCE_NOT_BOUND", "table.capabilities.igpv", "the binding declares no igpv payload digest", "P0")
    if not range_admits(parse_range(row.version_range), parse_version(R.CAPABILITY_VERSION)):
        raise IGPVError("PROVENANCE_VERSION_UNSUPPORTED", "table.capabilities.igpv.version_range",
                        f"{row.version_range} excludes IGPV/1 ({R.CAPABILITY_VERSION})", "P0")
    actual = sha256_hex(payload_bytes)
    if actual != row.payload_sha256:
        raise IGPVError("PAYLOAD_DIGEST_MISMATCH", "table.capabilities.igpv.payload_sha256", f"payload bytes hash to {actual}", "P0")
    return result


def _supplied(sources: Mapping[str, bytes]) -> dict[str, bytes]:
    """Only lowercase 64-hex keys count as supplied artifacts; anything else is ignored (never matched loosely)."""
    return {k: v for k, v in sources.items() if R.is_str(k) and R.HEX64.fullmatch(k) is not None and isinstance(v, bytes)}


def _check_supplied(artifact: R.Artifact, data: bytes, step: str) -> None:
    w = f"table.artifacts.{artifact.artifact_id}"
    actual = sha256_hex(data)
    if actual != artifact.artifact_sha256:
        raise IGPVError("SOURCE_ARTIFACT_MISMATCH", f"{w}.artifact_sha256", f"supplied bytes hash to {actual}", step)
    if len(data) != artifact.size:
        raise IGPVError("SOURCE_ARTIFACT_MISMATCH", f"{w}.size", f"supplied bytes are {len(data)} bytes, declared {artifact.size}", step)


def _sorted_ids(items, key):
    return sorted(items, key=lambda x: key(x).encode("utf-8", "surrogatepass"))


def verify(payload: str | bytes | ProvenancePayload, binding: ExtensionBinding, target: str | bytes, *,
           sources: Mapping[str, bytes] | None = None, limits: R.Limits = R.DEFAULT_LIMITS,
           baselines: Mapping[str, Baseline] = BASELINES, registry: CapabilityRegistry = DEFAULT_REGISTRY,
           ) -> tuple[VerificationResult, ProvenanceVerification]:
    sources = {} if sources is None else sources
    target_bytes = as_bytes(target)
    _preflight(target_bytes, sources, limits)                 # before any parser or serializer touches anything
    payload_bytes = _payload_bytes(payload)                   # an in-memory payload is serialized here: check() runs as part of it,
    if len(payload_bytes) > limits.max_payload_bytes:         # so its shape faults precede the payload-byte budget by necessity
        raise _limit("max_payload_bytes", f"payload {len(payload_bytes)} > {limits.max_payload_bytes}")
    igx_result = _p0(binding, payload_bytes, target_bytes, baselines, registry)                 # P0
    if isinstance(payload, ProvenancePayload):                                                  # P1
        p = payload
        for name in ("artifacts", "producers", "objects", "attributions", "derivations"):
            if len(getattr(p, name)) > limits.max_rows_per_table:
                raise _limit("max_rows_per_table", f"table.{name} has {len(getattr(p, name))} rows > {limits.max_rows_per_table}")
    else:
        p = ProvenancePayload.from_ison(payload_bytes, limits=limits)
    _graph_budget(p, _supplied(sources), limits)                                                # P1: origin graphs vs max_graph_bytes
    actual = sha256_hex(target_bytes)                                                           # P2
    if p.target_artifact_sha256 != actual:
        raise IGPVError("TARGET_BINDING_MISMATCH", "meta.igpv.target_artifact_sha256", f"target bytes hash to {actual}", "P2")
    if p.target_artifact_sha256 != binding.artifact_sha256:
        raise IGPVError("TARGET_BINDING_MISMATCH", "meta.igpv.target_artifact_sha256", f"binding names {binding.artifact_sha256}", "P2")
    if p.target_serialization_profile != binding.serialization_profile:
        raise IGPVError("TARGET_BINDING_MISMATCH", "meta.igpv.target_serialization_profile",
                        f"{p.target_serialization_profile} != binding {binding.serialization_profile}", "P2")
    check_references(p)                                                                         # P3
    arts = {a.artifact_id: a for a in p.artifacts}
    supplied = _supplied(sources)
    origin_ids = sorted({o.graph for o in p.objects if o.graph is not None}, key=lambda s: s.encode("utf-8"))
    origin_resolution: dict[str, str] = {}
    if igx_result.semantic_verification == "full":                                              # P4
        graph = load_graph(target_bytes, binding.serialization_profile, "target")
        for o in _sorted_ids(p.objects, lambda o: o.object_id):
            if o.graph is None:
                resolve_object(graph, o, f"table.objects.{o.object_id}")
        object_resolution = "full"
        for aid in origin_ids:
            a = arts[aid]
            data = supplied.get(a.artifact_sha256)
            if data is None:
                origin_resolution[aid] = "unverified"
                continue
            _check_supplied(a, data, "P4")
            g = load_graph(data, a.serialization_profile, f"table.artifacts.{aid}")
            for o in _sorted_ids(p.objects, lambda o: o.object_id):
                if o.graph == aid:
                    resolve_object(g, o, f"table.objects.{o.object_id}")
            origin_resolution[aid] = "verified"
    else:
        object_resolution = "skipped"
        origin_resolution = {aid: "unverified" for aid in origin_ids}
    locators: dict[str, str] = {}                                                               # P5
    for at in _sorted_ids(p.attributions, lambda a: a.attribution_id):
        if at.locator is None:
            continue
        a = arts[at.artifact_id]
        data = supplied.get(a.artifact_sha256)
        if data is not None:
            _check_supplied(a, data, "P5")
        locators[at.attribution_id] = locator_posture(at.locator, data, f"table.attributions.{at.attribution_id}")
    counts = {name: len(getattr(p, name)) for name in ("artifacts", "producers", "objects", "attributions", "derivations")}
    return igx_result, ProvenanceVerification(
        payload_sha256=sha256_hex(payload_bytes), binding_digest=binding.digest, target_artifact_sha256=p.target_artifact_sha256,
        object_resolution=object_resolution, origin_resolution=origin_resolution, locators=locators, counts=counts, steps_passed=STEPS)
```

```python
# isongraph-ext/src/isongraph_ext/igpv/bind.py
"""Producer path and trustworthy consumer path (18 §5.2): never return a binding whose provenance would not verify."""

from __future__ import annotations

from collections.abc import Mapping, Sequence

from ..igx.admit import DEFAULT_POLICY, AdmissionPolicy, AdmissionResult, admit
from ..igx.baselines import BASELINES, DEFAULT_BASELINE, Baseline
from ..igx.bind import bind as igx_bind
from ..igx.binding import Capability, Consumer, ExtensionBinding
from ..igx.digest import sha256_hex
from ..igx.registry import DEFAULT_REGISTRY, CapabilityRegistry
from ..igx.source import as_bytes
from ..igx.verify import VerificationResult
from . import records as R
from .errors import IGPVError
from .payload import ProvenancePayload
from .verify import ProvenanceVerification, verify


def bind(target: str | bytes, payload: ProvenancePayload, *, requirement: str = "REQUIRED", serialization_profile: str,
         producer_port: str, identity_claim: str, capabilities: Sequence[Capability] = (), consumer: Consumer | None = None,
         baseline: Baseline = DEFAULT_BASELINE, registry: CapabilityRegistry = DEFAULT_REGISTRY,
         policy: AdmissionPolicy = DEFAULT_POLICY, sources: Mapping[str, bytes] | None = None,
         limits: R.Limits = R.DEFAULT_LIMITS) -> ExtensionBinding:
    if requirement not in ("REQUIRED", "OPTIONAL"):
        raise IGPVError("MALFORMED_PROVENANCE", "table.capabilities.igpv.requirement", "REQUIRED or OPTIONAL", "P1")
    if any(c.extension == R.CAPABILITY for c in capabilities):
        raise IGPVError("MALFORMED_PROVENANCE", "table.capabilities.igpv", "bind() adds the igpv row itself; do not pass one", "P1")
    payload.check()
    data = as_bytes(target)
    actual = sha256_hex(data)
    if payload.target_artifact_sha256 != actual:
        raise IGPVError("TARGET_BINDING_MISMATCH", "meta.igpv.target_artifact_sha256", f"target bytes hash to {actual}", "P2")
    if payload.target_serialization_profile != serialization_profile:
        raise IGPVError("TARGET_BINDING_MISMATCH", "meta.igpv.target_serialization_profile",
                        f"{payload.target_serialization_profile} != {serialization_profile}", "P2")
    caps = tuple(capabilities) + (Capability(R.CAPABILITY, R.CAPABILITY_RANGE, requirement, payload.digest),)
    binding = igx_bind(data, serialization_profile=serialization_profile, producer_port=producer_port, identity_claim=identity_claim,
                       capabilities=caps, consumer=consumer, baseline=baseline, registry=registry, policy=policy)
    baselines = {**BASELINES, baseline.core_commit: baseline}
    verify(payload, binding, data, sources=sources, limits=limits, baselines=baselines, registry=registry)
    return binding


def verify_and_admit(payload: str | bytes | ProvenancePayload, binding: ExtensionBinding, target: str | bytes, *,
                     sources: Mapping[str, bytes] | None = None, limits: R.Limits = R.DEFAULT_LIMITS,
                     baselines: Mapping[str, Baseline] = BASELINES, registry: CapabilityRegistry = DEFAULT_REGISTRY,
                     policy: AdmissionPolicy = DEFAULT_POLICY) -> tuple[VerificationResult, ProvenanceVerification, AdmissionResult]:
    result, prov = verify(payload, binding, target, sources=sources, limits=limits, baselines=baselines, registry=registry)
    decision = admit(binding, result, policy=policy, baselines=baselines)       # 18 §4.2: the binding's decision, unchanged
    return result, prov, decision
```

```python
# isongraph-ext/src/isongraph_ext/igpv/lineage.py
"""Bounded, read-only lineage queries across payloads (18 §5.2). Never merges, ranks, loads graphs or touches bytes."""

from __future__ import annotations

from collections.abc import Iterable

from . import records as R
from .errors import IGPVError
from .payload import ProvenancePayload, address_key, check_references

Address = R.NodeRef | R.EdgeAddress | None


def _key(obj: Address) -> tuple:
    if obj is None:
        return ("graph",)
    if isinstance(obj, R.NodeRef):
        return ("node", obj)
    return ("edge", obj)


def _bytes(s: str) -> bytes:
    return s.encode("utf-8", "surrogatepass")


class _Index:
    def __init__(self, p: ProvenancePayload) -> None:
        self.payload = p
        self.arts = {a.artifact_id: a for a in p.artifacts}
        self.prods = {x.producer_id: x for x in p.producers}
        self.objs = {o.object_id: o for o in p.objects}
        self.by_address = {address_key(o): o for o in p.objects if o.graph is None}
        self.derivations: dict[str, list[R.Derivation]] = {}
        for d in sorted(p.derivations, key=lambda d: _bytes(d.derivation_id)):
            self.derivations.setdefault(d.derived_object_id, []).append(d)
        self.attributions = sorted(p.attributions, key=lambda a: _bytes(a.attribution_id))

    def graph_sha(self, o: R.GraphObject) -> str:
        return self.payload.target_artifact_sha256 if o.graph is None else self.arts[o.graph].artifact_sha256

    @staticmethod
    def address(o: R.GraphObject) -> Address:
        return o.node if o.kind == "node" else o.edge if o.kind == "edge" else None


class Lineage:
    def __init__(self, payloads: Iterable[ProvenancePayload], *, limits: R.Limits = R.DEFAULT_LIMITS) -> None:
        self._limits = limits
        self._by_target: dict[str, _Index] = {}
        for p in payloads:
            p.check()
            check_references(p)
            if p.target_artifact_sha256 in self._by_target:
                raise IGPVError("LINEAGE_AMBIGUOUS", "meta.igpv.target_artifact_sha256",
                                f"two payloads describe {p.target_artifact_sha256}; merging is not an IGPV/1 operation", "L")
            self._by_target[p.target_artifact_sha256] = _Index(p)

    def origins_of(self, graph_sha256: str, obj: Address) -> tuple[R.LineageStep, ...]:
        lim = self._limits
        steps: list[R.LineageStep] = []
        visits = 0
        # stack entries: (graph_sha, address, depth, path, kind, producer) — DFS preorder in canonical record order
        stack: list[tuple] = [(graph_sha256, obj, 0, frozenset(), None, None)]
        while stack:
            sha, address, depth, path, kind, producer = stack.pop()
            key = (sha, _key(address))
            if key in path:
                raise IGPVError("DERIVATION_CYCLE", "table.derivations", f"{address!r} in {sha} recurs on the lineage path", "L")
            visits += 1
            if visits > lim.max_lineage_visits:
                raise IGPVError("LINEAGE_BUDGET_EXCEEDED", "max_lineage_visits", f"more than {lim.max_lineage_visits} objects visited", "L")
            if depth > lim.max_lineage_depth:
                raise IGPVError("LINEAGE_BUDGET_EXCEEDED", "max_lineage_depth", f"deeper than {lim.max_lineage_depth}", "L")
            if kind is not None:
                steps.append(R.LineageStep(depth, sha, address, kind, producer, None))
            idx = self._by_target.get(sha)
            if idx is None:
                steps.append(R.LineageStep(depth, sha, address, None, None, "payload_unavailable"))
                continue
            o = idx.by_address.get(_key(address))
            rows = idx.derivations.get(o.object_id, []) if o is not None else []
            if not rows:
                steps.append(R.LineageStep(depth, sha, address, None, None, "no_derivation_record"))
                continue
            for d in reversed(rows):
                origin = idx.objs[d.origin_object_id]
                stack.append((idx.graph_sha(origin), idx.address(origin), depth + 1, path | {key}, d.kind, idx.prods[d.producer_id]))
        return tuple(steps)

    def attributions_of(self, graph_sha256: str, obj: Address) -> tuple[R.Attribution, ...]:
        idx = self._by_target.get(graph_sha256)
        if idx is None:
            return ()
        o = idx.by_address.get(_key(obj))
        if o is None:
            return ()
        return tuple(a for a in idx.attributions if a.object_id == o.object_id)
```

```python
"""CLI (18 §5.3): verify -> 0/1/2, origins -> 0/1, 64 on usage. Output is ISON so receipts can quote it verbatim."""

from __future__ import annotations

import argparse
import re
import sys
from collections.abc import Mapping
from pathlib import Path

from ison_parser import Block, Document, dumps_canonical

from ..igx.admit import AdmissionPolicy
from ..igx.binding import ExtensionBinding
from ..igx.errors import IGXError
from . import records as R
from .bind import verify_and_admit
from .errors import IGPVError
from .lineage import Lineage
from .payload import ProvenancePayload

EX_USAGE = 64
LINEAGE_FIELDS = ("depth", "graph_sha256", "object_kind", "node_type", "node_id", "source_type", "source_id", "relation",
                  "target_type", "target_id", "ordinal", "derivation_kind", "producer_implementation", "producer_version", "terminal")
_DECIMAL = re.compile(r"-?(0|[1-9][0-9]*)")


class _Parser(argparse.ArgumentParser):
    def error(self, message: str) -> None:  # type: ignore[override]
        self.print_usage(sys.stderr)
        sys.stderr.write(f"error: {message}\n")
        raise SystemExit(EX_USAGE)


def _utf8_streams() -> None:
    for stream in (sys.stdout, sys.stderr):
        reconfigure = getattr(stream, "reconfigure", None)
        if reconfigure is not None:
            reconfigure(encoding="utf-8", errors="strict", newline="\n")


def _meta(name: str, row: dict) -> Block:
    return Block("meta", name, list(row), [row])


def _emit(blocks: list[Block]) -> None:
    """One response = one canonical ISON document, written once (block separation and order are the writer's)."""
    sys.stdout.write(dumps_canonical(Document(blocks=blocks)) + "\n")


def _error(e: IGXError | IGPVError) -> int:
    _emit([_meta("error", {"code": e.code, "field": e.field, "detail": e.detail, "step": e.step})])
    return 1


def _joined(items) -> str | None:
    items = list(items)
    return ",".join(items) if items else None


def _kv(m: Mapping) -> str | None:
    return _joined(f"{k}={v}" for k, v in m.items())


def _usage(ap: _Parser, message: str) -> int:
    ap.print_usage(sys.stderr)
    sys.stderr.write(f"error: {message}\n")
    return EX_USAGE


def _id_token(s: str) -> int | str:
    """`1` is INT64, `"1"` is STRING, anything else is a STRING as typed (shell quoting is the caller's)."""
    if _DECIMAL.fullmatch(s):
        v = int(s)
        if not (R.INT64_MIN <= v <= R.INT64_MAX):
            raise ValueError(f"integer id outside INT64: {s}")
        return v
    if len(s) >= 2 and s[0] == s[-1] == '"':
        return s[1:-1]
    return s


def _address_row(obj) -> dict:
    row: dict = {k: None for k in LINEAGE_FIELDS}
    if obj is None:
        row["object_kind"] = "graph"
    elif isinstance(obj, R.NodeRef):
        row.update({"object_kind": "node", "node_type": obj.node_type, "node_id": obj.node_id})
    else:
        row.update({"object_kind": "edge", "source_type": obj.source.node_type, "source_id": obj.source.node_id, "relation": obj.relation,
                    "target_type": obj.target.node_type, "target_id": obj.target.node_id, "ordinal": obj.ordinal})
    return row


def main(argv: list[str]) -> int:
    _utf8_streams()
    ap = _Parser(prog="python -m isongraph_ext.igpv")
    sub = ap.add_subparsers(dest="cmd", required=True)
    v = sub.add_parser("verify")
    v.add_argument("graph", type=Path)
    v.add_argument("binding", type=Path)
    v.add_argument("payload", type=Path)
    v.add_argument("--source", action="append", default=[], metavar="SHA256=PATH")
    v.add_argument("--allow-retired", action="store_true")
    o = sub.add_parser("origins")
    o.add_argument("graph_sha256")
    o.add_argument("payloads", nargs="+", type=Path)
    o.add_argument("--node", nargs=2, metavar=("TYPE", "ID"))
    o.add_argument("--edge", nargs="+", metavar="PART")
    a = ap.parse_args(argv)
    if a.cmd == "verify":
        return _verify(ap, a)
    return _origins(ap, a)


def _verify(ap: _Parser, a) -> int:
    sources: dict[str, bytes] = {}
    try:
        graph, binding_bytes, payload_bytes = a.graph.read_bytes(), a.binding.read_bytes(), a.payload.read_bytes()
        for item in a.source:
            key, sep, path = item.partition("=")
            if not sep or R.HEX64.fullmatch(key) is None:
                return _usage(ap, f"--source needs SHA256=PATH with lowercase hex: {item!r}")
            sources[key] = Path(path).read_bytes()
    except OSError as e:
        return _usage(ap, str(e))
    try:
        binding = ExtensionBinding.from_ison(binding_bytes)
        result, prov, decision = verify_and_admit(payload_bytes, binding, graph, sources=sources,
                                                  policy=AdmissionPolicy(allow_retired_baselines=a.allow_retired))
    except (IGXError, IGPVError) as e:
        return _error(e)
    _emit([
        _meta("verification", {
            "binding_digest": result.binding_digest, "semantic_verification": result.semantic_verification,
            "source_canonical": result.source_canonical, "source_semantic_eligibility": result.source_semantic_eligibility,
            "source_semantic_reason": result.source_semantic_reason, "steps_passed": _joined(result.steps_passed),
            "unrecognized_optional": _joined(c.extension for c in result.unrecognized_optional)}),
        _meta("provenance", {
            "payload_sha256": prov.payload_sha256, "binding_digest": prov.binding_digest, "target_artifact_sha256": prov.target_artifact_sha256,
            "object_resolution": prov.object_resolution, "origin_resolution": _kv(prov.origin_resolution), "locators": _kv(prov.locators),
            "counts": _kv(prov.counts), "steps_passed": _joined(prov.steps_passed)}),
        _meta("admission", {"admitted": decision.admitted, "code": decision.code, "step": decision.step,
                            "receipt_posture": decision.receipt_posture, "reasons": _joined(decision.reasons)}),
    ])
    return 0 if decision.admitted else 2


def _origins(ap: _Parser, a) -> int:
    if R.HEX64.fullmatch(a.graph_sha256) is None:
        return _usage(ap, "graph_sha256 must be 64 lowercase hex")
    if a.node and a.edge:
        return _usage(ap, "--node and --edge are exclusive")
    try:
        if a.node:
            obj = R.NodeRef(a.node[0], _id_token(a.node[1]))
        elif a.edge:
            if len(a.edge) not in (5, 6):
                return _usage(ap, "--edge needs STYPE SID REL TTYPE TID [ORDINAL]")
            st, sid, rel, tt, tid = a.edge[:5]
            obj = R.EdgeAddress(R.NodeRef(st, _id_token(sid)), rel, R.NodeRef(tt, _id_token(tid)), a.edge[5] if len(a.edge) == 6 else "0")
        else:
            obj = None
        texts = [p.read_bytes() for p in a.payloads]
    except (OSError, ValueError) as e:
        return _usage(ap, str(e))
    try:
        steps = Lineage([ProvenancePayload.from_ison(t) for t in texts]).origins_of(a.graph_sha256, obj)
    except IGPVError as e:
        return _error(e)
    rows = []
    for s in steps:
        row = _address_row(s.obj)
        row.update({"depth": s.depth, "graph_sha256": s.graph_sha256, "derivation_kind": s.kind, "terminal": s.terminal,
                    "producer_implementation": None if s.producer is None else s.producer.implementation,
                    "producer_version": None if s.producer is None else s.producer.version})
        rows.append(row)
    _emit([Block("table", "lineage", list(LINEAGE_FIELDS), rows)])
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
```

```python
def _details(detail: str | None) -> dict[str, list[str]]:
    d: dict[str, list[str]] = {}
    for item in (detail or "").split(";"):
        if ":" in item:
            k, v = item.split(":", 1)
            d.setdefault(k.strip(), []).append(v.strip())
    return d


def check(id_, target_rel, binding_text, payload_texts, sources, expect, step, detail) -> None:
    d = _details(detail)
    limits = Limits(**{k: int(v) for item in d.get("limits", []) for k, v in [item.split("=")]})
    src = {} if sources == "none" else {name: SOURCES[name] for name in sources.split("+")}
    tokens = expect.split("+")
    if id_.startswith("lineage/"):
        payloads = [ProvenancePayload.from_ison(t) for t in payload_texts]
        obj = next(o for o in payloads[0].objects if o.object_id == d["query"][0])
        addr = obj.node if obj.kind == "node" else obj.edge if obj.kind == "edge" else None
        try:
            steps = Lineage(payloads, limits=limits).origins_of(payloads[0].target_artifact_sha256, addr)
        except IGPVError as e:
            assert (e.code, e.step) == (tokens[0][len("lineage:"):], step), (id_, e)
            return
        assert "lineage_ok" in tokens, id_
        for n in d.get("origins", []):
            assert sum(1 for s in steps if s.kind) == int(n), (id_, steps)
        for t in d.get("terminal", []):
            assert steps[-1].terminal == t, (id_, steps)
        return
    target = GRAPHS[target_rel]
    try:
        result, prov, decision = verify_and_admit(payload_texts.encode("utf-8"), ExtensionBinding.from_ison(binding_text), target,
                                                  sources=src, limits=limits, baselines=B)
    except (IGXError, IGPVError) as e:
        s = "P0" if isinstance(e, IGXError) else e.step
        assert tokens[0].startswith("reject:") and (e.code, s) == (tokens[0][len("reject:"):], step), (id_, e)
        return
    assert "verify_ok" in tokens, id_
    for item in d.get("locator", []):
        aid, posture = item.split("=")
        assert prov.locators[aid] == posture, (id_, prov.locators)
    for item in d.get("origin", []):
        aid, status = item.split("=")
        assert prov.origin_resolution[aid] == status, (id_, prov.origin_resolution)
    assert prov.object_resolution == ("skipped" if "skipped" in d.get("resolution", []) else "full"), id_
    for t in tokens:
        if t == "admit_ok":
            assert decision.admitted, (id_, decision)
        elif t.startswith("admit:"):
            assert (decision.admitted, decision.code, decision.step) == (False, t[len("admit:"):], step), (id_, decision)


def write_all() -> None:
    for sub in ("graphs/synthetic", "bindings", "payloads", "sources"):
        shutil.rmtree(VEC / sub, ignore_errors=True)
        (VEC / sub).mkdir(parents=True)
    for rel, data in GRAPHS.items():
        (VEC / "graphs" / rel).write_bytes(data)
    for name, data in SOURCES.items():
        (VEC / "sources" / name).write_bytes(data)
    rows = []
    for id_, target_rel, binding_text, payload_texts, sources, expect, step, detail in VECTORS:
        if binding_text is None:
            binding_text = bind_text(payload_texts, GRAPHS[target_rel])
        check(id_, target_rel, binding_text, payload_texts, sources, expect, step, detail)
        slug = id_.replace("/", "--")
        if isinstance(payload_texts, list):
            names = []
            for i, t in enumerate(payload_texts):
                name = f"{slug}-{i}.igpv.ison"
                (VEC / "payloads" / name).write_bytes(t.encode("utf-8"))
                names.append(name)
            payload_name, binding_name = "+".join(names), "none"
        else:
            payload_name, binding_name = f"{slug}.igpv.ison", f"{slug}.igx.ison"
            (VEC / "payloads" / payload_name).write_bytes(payload_texts.encode("utf-8"))
            (VEC / "bindings" / binding_name).write_bytes(binding_text.encode("utf-8"))
        rows.append({"id": id_, "target": target_rel, "binding": binding_name, "payload": payload_name, "sources": sources,
                     "expect": expect, "step": step, "detail": detail})
    doc = Document(blocks=[Block("meta", "corpus", ["provenance_spec", "version"], [{"provenance_spec": "IGPV/1", "version": 1}]),
                           Block("table", "vectors", ["id", "target", "binding", "payload", "sources", "expect", "step", "detail"], rows)])
    (VEC / "manifest.ison").write_bytes((dumps_canonical(doc) + "\n").encode("utf-8"))
    print(f"wrote {len(rows)} vectors")


if __name__ == "__main__":
    write_all()
```
