"""The one exception type IGAR/1 raises (25 §3). Verification errors of igx/igdm/igdc are recorded INSIDE receipts, never raised by issue()."""
from __future__ import annotations


class IGARError(Exception):
    def __init__(self, code: str, field: str, detail: str, step: str) -> None:
        super().__init__(f"{step} {code} {field}: {detail}")
        self.code, self.field, self.detail, self.step = code, field, detail, step

"""IGAR/1 receipt model, canonical wire and checks R0-R2 (25 §2-§3). The receipt digest is external to the bytes (25 §2).

Reading goes through `strict_ison.read_canonical`: byte budget before parsing, canonical form, exact blocks, unique columns, no
summary / typed columns, scalar cells. Passing R0-R2 means a receipt is well-formed and self-consistent. It does NOT mean it is true,
bound to any bytes, delivered by a trusted channel, or acceptable to any policy: those are R3, R5 and R4 (`check.py`, `trust.py`).
"""
from __future__ import annotations

import re
from dataclasses import dataclass

from ison_parser import Block, Document, dumps_canonical

from ..igdm.records import DEFAULT_LIMITS, Limits
from ..igx.digest import sha256_hex
from ..strict_ison import BlockShape, StrictError, read_canonical
from .errors import IGARError

RECEIPT_SPEC = "IGAR/1"
HEX64, HEX40 = re.compile(r"[0-9a-f]{64}"), re.compile(r"[0-9a-f]{40}")
FIELDS: dict[tuple[str, str], tuple[str, ...]] = {
    ("meta", "igar"): ("receipt_spec",),
    ("meta", "subject"): ("binding_sha256", "igdc_sha256", "igdm_payload_sha256", "target_artifact_sha256"),
    ("meta", "verifier"): ("verifier", "verifier_version", "igx_spec", "igdm_spec", "igdc_spec", "profile_sha256", "core_commit", "runtime_ison_graph",
                           "runtime_ison_py", "semantic_verification", "object_resolution"),
    ("meta", "result"): ("binding_schema", "payload_schema", "verification", "admission", "consumption"),
    ("meta", "failure"): ("check", "code", "field"),
    ("meta", "admission"): ("code", "step", "receipt_posture"),
    ("meta", "consumption"): ("code", "step", "field"),
    ("table", "reasons"): ("reason",),
    ("table", "unrecognized"): ("entry",),
}
SHAPES = tuple(BlockShape(k, n, f, k == "meta") for (k, n), f in FIELDS.items())
CHECKS = ("V0", "V1", "V2", "V3", "V4", "V5", "D0", "D1", "D2", "D3", "D4")
# failing check -> (binding_schema, payload_schema, verification)   (25 §2.2; written out, not computed)
STATUS: dict[str | None, tuple[str, str, str]] = {
    None: ("valid", "valid", "verified"),
    "V1": ("invalid", "not_evaluated", "not_evaluated"),
    "D1": ("valid", "invalid", "not_evaluated"),
    **{c: ("valid", "not_evaluated", "rejected") for c in ("V0", "V2", "V3", "V4", "V5", "D0")},
    **{c: ("valid", "valid", "rejected") for c in ("D2", "D3", "D4")},
}
ADMISSION_STEP = {"STALE_BINDING": "A1", "IDENTITY_CLAIM_INADMISSIBLE": "A2"}
POSTURES = ("OK", "CORE_LOSSY", "CORE_BLOCKING_DIVERGENCE", "INADMISSIBLE")
_V = {"binding_schema": ("valid", "invalid", "not_evaluated"), "payload_schema": ("valid", "invalid", "not_evaluated"),
      "verification": ("verified", "rejected", "not_evaluated"), "admission": ("admitted", "denied", "not_evaluated"),
      "consumption": ("derived", "failed", "absent")}


def _r1(field: str, detail: str) -> IGARError:
    return IGARError("MALFORMED_RECEIPT", field, detail, "R1")


def _r2(field: str, detail: str) -> IGARError:
    return IGARError("INCONSISTENT_RECEIPT", field, detail, "R2")


@dataclass(frozen=True)
class Receipt:
    subject: dict
    verifier: dict
    result: dict
    failure: dict
    admission: dict
    consumption: dict
    reasons: tuple[str, ...] = ()
    unrecognized: tuple[str, ...] = ()

    def to_document(self) -> Document:
        rows = {"igar": {"receipt_spec": RECEIPT_SPEC}, "subject": self.subject, "verifier": self.verifier, "result": self.result,
                "failure": self.failure, "admission": self.admission, "consumption": self.consumption}
        blocks = [Block("meta", n, list(FIELDS[("meta", n)]), [{f: r[f] for f in FIELDS[("meta", n)]}]) for n, r in rows.items()]
        blocks.append(Block("table", "reasons", ["reason"], [{"reason": x} for x in self.reasons]))
        blocks.append(Block("table", "unrecognized", ["entry"], [{"entry": x} for x in self.unrecognized]))
        return Document(blocks=blocks)

    def to_ison(self) -> str:
        return dumps_canonical(self.to_document())

    @property
    def digest(self) -> str:
        return sha256_hex(self.to_ison())

    @classmethod
    def from_ison(cls, data: bytes | str, *, limits: Limits = DEFAULT_LIMITS) -> "Receipt":
        """R0 and R1. The byte budget is `limits.max_payload_bytes`, applied before any parsing."""
        try:
            text, b = read_canonical(data, SHAPES, max_bytes=limits.max_payload_bytes, max_rows=limits.max_rows_per_table)
        except StrictError as e:
            step = "R0" if e.stage == "bytes" else "R1"
            if e.kind == "limit":
                raise IGARError("RECEIPT_LIMIT_EXCEEDED", "max_payload_bytes" if e.field == "size" else e.field, e.detail, step) from None
            raise IGARError("MALFORMED_RECEIPT", e.field, e.detail, step) from None
        one = lambda n: dict(b[("meta", n)][0])  # noqa: E731
        if one("igar")["receipt_spec"] != RECEIPT_SPEC:
            raise _r1("meta.igar.receipt_spec", f"must be {RECEIPT_SPEC}")
        r = cls(one("subject"), one("verifier"), one("result"), one("failure"), one("admission"), one("consumption"),
                tuple(x["reason"] for x in b[("table", "reasons")]), tuple(x["entry"] for x in b[("table", "unrecognized")]))
        r._check_shape()
        if r.to_ison() != text:                                  # a second, independent canonical check over the typed model
            raise IGARError("MALFORMED_RECEIPT", "canonical_form", "document is not the canonical writer's output for this receipt", "R0")
        return r

    def _check_shape(self) -> None:
        s, v, res, f, a = self.subject, self.verifier, self.result, self.failure, self.admission
        for col in ("binding_sha256", "igdm_payload_sha256", "target_artifact_sha256"):
            if not (isinstance(s[col], str) and HEX64.fullmatch(s[col])):
                raise _r1(f"meta.subject.{col}", "64 lowercase hex")
        if not (s["igdc_sha256"] is None or (isinstance(s["igdc_sha256"], str) and HEX64.fullmatch(s["igdc_sha256"]))):
            raise _r1("meta.subject.igdc_sha256", "null or 64 lowercase hex")
        for col, lit in (("verifier", "isongraph-ext"), ("igx_spec", "IGX/1"), ("igdm_spec", "IGDM/1"), ("igdc_spec", "IGDC/1")):
            if v[col] != lit:
                raise _r1(f"meta.verifier.{col}", f"must be {lit}")
        if not (isinstance(v["profile_sha256"], str) and HEX64.fullmatch(v["profile_sha256"])):
            raise _r1("meta.verifier.profile_sha256", "64 lowercase hex (never null: the issuer always ran under some profile)")
        for col in ("verifier_version", "runtime_ison_graph", "runtime_ison_py"):
            if not (isinstance(v[col], str) and v[col]):
                raise _r1(f"meta.verifier.{col}", "non-empty string")
        if not (v["core_commit"] is None or (isinstance(v["core_commit"], str) and HEX40.fullmatch(v["core_commit"]))):
            raise _r1("meta.verifier.core_commit", "null or 40 lowercase hex")
        if v["semantic_verification"] not in ("full", "limited", None) or v["object_resolution"] not in ("full", "skipped", None):
            raise _r1("meta.verifier", "semantic_verification/object_resolution outside vocabulary")
        for col, allowed in _V.items():
            if res[col] not in allowed:
                raise _r1(f"meta.result.{col}", f"not one of {allowed}")
        if f["check"] is not None and f["check"] not in CHECKS:
            raise _r1("meta.failure.check", f"not one of {CHECKS}")
        for blk, cols in (("failure", ("code", "field")), ("consumption", ("code", "step", "field")), ("admission", ("code",))):
            for col in cols:
                x = getattr(self, blk)[col]
                if not (x is None or (isinstance(x, str) and x)):
                    raise _r1(f"meta.{blk}.{col}", "null or non-empty string")
        if a["step"] not in ("A1", "A2", None) or a["receipt_posture"] not in (*POSTURES, None):
            raise _r1("meta.admission", "step/receipt_posture outside vocabulary")
        if a["code"] is not None and a["code"] not in ADMISSION_STEP:
            raise _r1("meta.admission.code", f"not one of {tuple(ADMISSION_STEP)}")
        for x in (*self.reasons, *self.unrecognized):
            if not (isinstance(x, str) and x):
                raise _r1("table", "rows must be non-empty strings")

    def check_consistency(self) -> None:
        """R2: the statuses are exactly what the failure record implies (25 §2.2)."""
        res, f, a, c, v, s = self.result, self.failure, self.admission, self.consumption, self.verifier, self.subject
        chk = f["check"]
        if chk not in STATUS:
            raise _r2("meta.failure.check", "unknown check")
        if (res["binding_schema"], res["payload_schema"], res["verification"]) != STATUS[chk]:
            raise _r2("meta.result", f"statuses do not match failure.check={chk!r}: expected {STATUS[chk]}")
        if (f["code"] is None) != (chk is None) or (f["field"] is None) != (chk is None):
            raise _r2("meta.failure", "code and field are non-null exactly when check is non-null")
        verified = chk is None
        if (v["core_commit"] is None) != (chk == "V1"):
            raise _r2("meta.verifier.core_commit", "null exactly when the binding failed to parse (check V1)")
        if (v["semantic_verification"] is None) == verified or (v["object_resolution"] is None) == verified:
            raise _r2("meta.verifier", "semantic_verification/object_resolution are non-null exactly when verification passed")
        if not verified:
            if res["admission"] != "not_evaluated" or any(a[k] is not None for k in a) or self.reasons or self.unrecognized:
                raise _r2("meta.admission", "admission, reasons and unrecognized are empty unless verification passed")
        else:
            if (a["code"] is None) != (a["step"] is None) or (a["code"] is not None and ADMISSION_STEP[a["code"]] != a["step"]):
                raise _r2("meta.admission", "code and step must be a matching pair or both null")
            denied = a["code"] is not None
            if res["admission"] != ("denied" if denied else "admitted"):
                raise _r2("meta.result.admission", "must be denied exactly when meta.admission.code is set")
            if a["receipt_posture"] is None or (denied != (a["receipt_posture"] == "INADMISSIBLE")):
                raise _r2("meta.admission.receipt_posture", "INADMISSIBLE exactly when denied")
        igdc, ccode = s["igdc_sha256"], c["code"]
        if (c["step"] is None) != (ccode is None) or (c["field"] is None) != (ccode is None):
            raise _r2("meta.consumption", "code, step and field are all null or all set")
        expected = "derived" if igdc is not None else "failed" if ccode is not None else "absent"
        if igdc is not None and ccode is not None:
            raise _r2("meta.consumption", "a hint set cannot be both derived and failed")
        if res["consumption"] != expected:
            raise _r2("meta.result.consumption", f"expected {expected}")
        if expected != "absent" and not verified:
            raise _r2("meta.result.consumption", "derivation needs a passed verification")
        if expected == "derived" and v["object_resolution"] != "full":
            raise _r2("meta.verifier.object_resolution", "an IGDC/1 hint set needs full object resolution")


def passed(r: Receipt) -> bool:
    return (r.result["binding_schema"], r.result["payload_schema"], r.result["verification"]) == ("valid", "valid", "verified")

# isongraph-ext/src/isongraph_ext/igar/profile.py
"""IGAR/1 verifier profile (25 §2.4): the EFFECTIVE verification configuration as a canonical ISON document.

Why it exists: the same target, binding and payload bytes can verify, or be denied, differently under different limits, baseline
registries, capability registries or admission policies. A receipt therefore records `profile_sha256`, and recomputation (R6) runs
under the profile that digest names - never under whatever defaults happen to be installed. The profile carries every normative
input explicitly (no digest-only indirection): limits, the capability registry, the admission policy, and for each baseline every
field verification reads (commit, format, runtime hashes, per-port conformance receipts, domains, retirement).
"""
from __future__ import annotations

import re
from collections.abc import Mapping
from dataclasses import dataclass
from functools import lru_cache
from types import MappingProxyType

from ison_parser import Block, Document, dumps_canonical

from ..igdm.records import DEFAULT_LIMITS, Limits
from ..igx.admit import DEFAULT_POLICY, AdmissionPolicy
from ..igx.baselines import BASELINES, PORTS, Baseline
from ..igx.baselines import Receipt as BaselineReceipt
from ..igx.digest import sha256_hex
from ..igx.registry import DEFAULT_REGISTRY
from ..strict_ison import BlockShape, StrictError, read_canonical
from .errors import IGARError

PROFILE_SPEC = "IGAR-PROFILE/1"
LIMIT_COLS = ("max_payload_bytes", "max_graph_bytes", "max_rows_per_table", "max_group_depth")
BASELINE_COLS = ("core_commit", "core_format", "core_version", "core_semantic_spec", "corpus_version", "corpus_commit", "corpus_digest",
                 "core_runtime_dist", "core_runtime_dist_version", "canonical_writer_dist", "canonical_writer_dist_version", "retired", "adopted")
RECEIPT_COLS = ("baseline", "port", "sha256", "port_version", "posture", "receipt_core_commit", "core_format", "core_version", "corpus_commit",
                "corpus_digest", "corpus_version", "semantic_spec")
DOMAIN_COLS = ("baseline", "port", "tag")
RUNTIME_COLS = ("baseline", "module", "sha256")
SHAPES = (BlockShape("meta", "profile", ("profile_spec",), True), BlockShape("meta", "limits", LIMIT_COLS, True),
          BlockShape("meta", "admission_policy", ("allow_retired_baselines",), True), BlockShape("table", "accept_lossy_domains", ("domain",)),
          BlockShape("table", "registry", ("extension", "version")), BlockShape("table", "baselines", BASELINE_COLS),
          BlockShape("table", "baseline_receipts", RECEIPT_COLS), BlockShape("table", "baseline_domains", DOMAIN_COLS),
          BlockShape("table", "baseline_runtime", RUNTIME_COLS))
_HEX = re.compile(r"[0-9a-f]{64}")


def _bad(field: str, detail: str) -> IGARError:
    return IGARError("MALFORMED_PROFILE", field, detail, "R6")


def _tok(b: bool) -> str:
    return "true" if b else "false"


@dataclass(frozen=True)
class Profile:
    limits: Limits
    baselines: Mapping[str, Baseline]
    registry: Mapping[str, str]
    policy: AdmissionPolicy

    @classmethod
    def default(cls) -> "Profile":
        """What this installed package verifies with when the caller names nothing. Recorded in every receipt it issues."""
        return _default()

    def to_document(self) -> Document:
        lim = {c: getattr(self.limits, c) for c in LIMIT_COLS}
        bases, recs, doms, runs = [], [], [], []
        for key in sorted(self.baselines):
            b = self.baselines[key]
            bases.append({"core_commit": b.core_commit, "core_format": b.core_format, "core_version": b.core_version, "core_semantic_spec": b.core_semantic_spec,
                          "corpus_version": b.corpus_version, "corpus_commit": b.corpus_commit, "corpus_digest": b.corpus_digest,
                          "core_runtime_dist": b.core_runtime_dist[0], "core_runtime_dist_version": b.core_runtime_dist[1],
                          "canonical_writer_dist": b.canonical_writer_dist[0], "canonical_writer_dist_version": b.canonical_writer_dist[1],
                          "retired": _tok(b.retired), "adopted": b.adopted})
            for port, r in sorted(b.receipts.items()):
                recs.append({"baseline": key, "port": port, "sha256": r.sha256, "port_version": r.port_version, "posture": r.posture,
                             "receipt_core_commit": r.core_commit, "core_format": r.core_format, "core_version": r.core_version, "corpus_commit": r.corpus_commit,
                             "corpus_digest": r.corpus_digest, "corpus_version": r.corpus_version, "semantic_spec": r.semantic_spec})
            for port, tags in sorted(b.domains.items()):
                doms.extend({"baseline": key, "port": port, "tag": t} for t in sorted(tags))
            runs.extend({"baseline": key, "module": m, "sha256": h} for m, h in sorted(b.core_runtime.items()))
        return Document(blocks=[
            Block("meta", "profile", ["profile_spec"], [{"profile_spec": PROFILE_SPEC}]), Block("meta", "limits", list(LIMIT_COLS), [lim]),
            Block("meta", "admission_policy", ["allow_retired_baselines"], [{"allow_retired_baselines": _tok(self.policy.allow_retired_baselines)}]),
            Block("table", "accept_lossy_domains", ["domain"], [{"domain": d} for d in sorted(self.policy.accept_lossy_domains)]),
            Block("table", "registry", ["extension", "version"], [{"extension": k, "version": v} for k, v in sorted(self.registry.items())]),
            Block("table", "baselines", list(BASELINE_COLS), bases), Block("table", "baseline_receipts", list(RECEIPT_COLS), recs),
            Block("table", "baseline_domains", list(DOMAIN_COLS), doms), Block("table", "baseline_runtime", list(RUNTIME_COLS), runs)])

    def to_ison(self) -> str:
        return dumps_canonical(self.to_document())

    @property
    def digest(self) -> str:
        return sha256_hex(self.to_ison())

    @classmethod
    def from_ison(cls, data: bytes | str, *, max_bytes: int = DEFAULT_LIMITS.max_payload_bytes, max_rows: int = DEFAULT_LIMITS.max_rows_per_table) -> "Profile":
        """Read a profile document: byte budget first, canonical, exact blocks, then typed rebuild. Raises MALFORMED_PROFILE / RECEIPT_LIMIT_EXCEEDED."""
        try:
            text, b = read_canonical(data, SHAPES, max_bytes=max_bytes, max_rows=max_rows)
        except StrictError as e:
            if e.kind == "limit":
                raise IGARError("RECEIPT_LIMIT_EXCEEDED", "max_payload_bytes" if e.field == "size" else e.field, e.detail, "R6") from None
            raise _bad(e.field, e.detail) from None
        row = lambda n: b[("meta", n)][0]  # noqa: E731
        if row("profile")["profile_spec"] != PROFILE_SPEC:

# isongraph-ext/src/isongraph_ext/igar/issue.py
"""Issue a receipt (25 §5): wraps the existing verification path; a rejection IS the content, never an exception.

Python reference only: IGAR/1 receipts are issued by this package. A receipt from any other producer is unqualified (25 §8).
"""
from __future__ import annotations

from importlib import metadata

from .. import __version__
from ..igdc.derive import consume
from ..igdc.errors import IGDCError
from ..igdm.errors import IGDMError
from ..igdm.verify import verify as igdm_verify
from ..igx.admit import admit
from ..igx.binding import ExtensionBinding
from ..igx.digest import sha256_hex
from ..igx.errors import IGXError
from .profile import Profile
from .receipt import STATUS, Receipt


def issue(binding: bytes, payload: bytes, target: bytes, *, igdc: bool = False, profile: Profile | None = None,
          verifier_version: str | None = None) -> Receipt:
    """Verify under `profile` (default: `Profile.default()`) and record the outcome AND the profile's digest.

    There are deliberately no separate limits/baselines/registry/policy arguments: the profile is the single statement of the
    effective configuration, so a receipt can never describe a run that used different settings than it names.
    """
    for name, v in (("binding", binding), ("payload", payload), ("target", target)):
        if not isinstance(v, (bytes, bytearray)):
            raise TypeError(f"{name} must be bytes (the verifier hashes exactly what it is given), got {type(v).__name__}")
    binding, payload, target = bytes(binding), bytes(payload), bytes(target)
    profile = profile or Profile.default()
    parsed = result = dm = adm = None
    failure: tuple[str, str, str] | None = None
    try:
        parsed = ExtensionBinding.from_ison(binding)                           # IGX/1 V1 (parse)
        result, dm = igdm_verify(payload, parsed, target, limits=profile.limits, baselines=profile.baselines, registry=profile.registry)
        adm = admit(parsed, result, policy=profile.policy, baselines=profile.baselines)
    except (IGXError, IGDMError) as e:
        failure = (e.step, e.code, e.field)
    chk = failure[0] if failure else None
    if chk not in STATUS:
        raise RuntimeError(f"verification reported an unknown step {chk!r}; IGAR/1 §2.2 must be extended by a new spec version")
    binding_schema, payload_schema, verification = STATUS[chk]
    ok = failure is None
    consumption, igdc_digest, c_code = "absent", None, (None, None, None)
    if igdc and ok:
        try:
            igdc_digest = consume(payload, parsed, target, limits=profile.limits, baselines=profile.baselines, registry=profile.registry).digest
            consumption = "derived"
        except IGDCError as e:
            consumption, c_code = "failed", (e.code, e.step, e.field)
    return Receipt(
        subject={"binding_sha256": sha256_hex(binding), "igdc_sha256": igdc_digest, "igdm_payload_sha256": sha256_hex(payload),
                 "target_artifact_sha256": sha256_hex(target)},
        verifier={"verifier": "isongraph-ext", "verifier_version": verifier_version or __version__, "igx_spec": "IGX/1", "igdm_spec": "IGDM/1",
                  "igdc_spec": "IGDC/1", "profile_sha256": profile.digest, "core_commit": parsed.core_commit if parsed else None,
                  "runtime_ison_graph": metadata.version("ison-graph"), "runtime_ison_py": metadata.version("ison-py"),
                  "semantic_verification": result.semantic_verification if ok else None, "object_resolution": dm.object_resolution if ok else None},
        result={"binding_schema": binding_schema, "payload_schema": payload_schema, "verification": verification,
                "admission": ("admitted" if adm.admitted else "denied") if ok else "not_evaluated", "consumption": consumption},
        failure={"check": chk, "code": failure[1] if failure else None, "field": failure[2] if failure else None},
        admission={"code": adm.code if ok else None, "step": adm.step if ok else None, "receipt_posture": adm.receipt_posture if ok else None},
        consumption={"code": c_code[0], "step": c_code[1], "field": c_code[2]},
        reasons=tuple(adm.reasons) if ok else (), unrecognized=tuple(dm.unrecognized_roles) if ok else ())

"""The receipt operations that need no trust decision (25 §3, §4): inspect (R0-R2), subject (R3), recompute (R6, Python only).

Nothing here says a receipt is TRUSTED. `inspect_receipt` says it is well-formed and consistent; `check_subject` says it is about
exactly these bytes; `recompute` says a Python verifier reproduces it. Authorizing use of hints is `trust.authorize_hints`.
"""
from __future__ import annotations

from importlib import metadata

from .. import __version__
from ..igdm.records import DEFAULT_LIMITS, Limits
from ..igx.digest import sha256_hex
from .errors import IGARError
from .issue import issue
from .profile import Profile
from .receipt import Receipt


def inspect_receipt(data: bytes | str, *, limits: Limits = DEFAULT_LIMITS) -> Receipt:
    """R0 (bytes, budget, canonical form), R1 (shape), R2 (consistency). Needs no target, no binding, no payload, no policy.

    A receipt that passes is a well-formed, self-consistent CLAIM. It is not evidence that the claim is true or that anyone
    trustworthy made it.
    """
    r = Receipt.from_ison(data, limits=limits)
    r.check_consistency()
    return r


def _digest_of(name: str, data: object) -> str:
    if not isinstance(data, (bytes, bytearray)):
        raise TypeError(f"{name} must be bytes, got {type(data).__name__}")
    return sha256_hex(bytes(data))


def check_subject(receipt: Receipt, *, target: bytes, binding: bytes, payload: bytes, igdc: bytes | None) -> None:
    """R3: the receipt is about EXACTLY these bytes. Every input is a required keyword: an omitted one is a TypeError, never a skipped check.

    `igdc` is `None` only for "I hold no hint document"; then the receipt must name none either. A receipt that names a hint
    document the caller did not supply, or a caller who supplies one the receipt does not name, is a mismatch in both directions.
    """
    s = receipt.subject
    for field, name, data in (("target_artifact_sha256", "target", target), ("binding_sha256", "binding", binding), ("igdm_payload_sha256", "payload", payload)):
        if _digest_of(name, data) != s[field]:
            raise IGARError("RECEIPT_SUBJECT_MISMATCH", f"meta.subject.{field}", f"the receipt is not about the {name} bytes supplied", "R3")
    if igdc is None:
        if s["igdc_sha256"] is not None:
            raise IGARError("RECEIPT_SUBJECT_MISMATCH", "meta.subject.igdc_sha256", "the receipt names a hint document the caller did not supply", "R3")
    elif s["igdc_sha256"] is None or _digest_of("igdc", igdc) != s["igdc_sha256"]:
        raise IGARError("RECEIPT_SUBJECT_MISMATCH", "meta.subject.igdc_sha256", "the receipt does not name the hint document supplied", "R3")


def _unavailable(field: str, detail: str) -> IGARError:
    return IGARError("RECOMPUTE_UNAVAILABLE", field, detail, "R6")


def recompute(receipt: bytes | str, binding: bytes, payload: bytes, target: bytes, *, profile: bytes | None, verifier_version: str | None = None) -> None:
    """R6 (Python reference only): re-run the verification path under the profile the receipt NAMES and require identical bytes.

    A receipt that cannot be re-run here (profile not supplied, profile differs, other verifier version, other runtime) raises
    `RECOMPUTE_UNAVAILABLE`: that says nothing against the receipt. Only `RECEIPT_RECOMPUTE_MISMATCH`, raised after every
    precondition holds, says the receipt is not what the verifier computes. The profile is never defaulted from the installed package.
    """
    r = inspect_receipt(receipt)
    v = r.verifier
    if profile is None:
        raise _unavailable("profile", f"the receipt was issued under profile {v['profile_sha256']}; supply that profile document")
    if _digest_of("profile", profile) != v["profile_sha256"]:
        raise _unavailable("meta.verifier.profile_sha256", "the supplied profile is not the one the receipt names")
    prof = Profile.from_ison(profile)
    if v["verifier_version"] != (verifier_version or __version__):
        raise _unavailable("meta.verifier.verifier_version", f"issued by isongraph-ext {v['verifier_version']}; this is {verifier_version or __version__}")
    for field, dist in (("runtime_ison_graph", "ison-graph"), ("runtime_ison_py", "ison-py")):
        if v[field] != metadata.version(dist):
            raise _unavailable(f"meta.verifier.{field}", f"issued under {dist} {v[field]}; installed {metadata.version(dist)}")
    again = issue(binding, payload, target, igdc=(r.subject["igdc_sha256"] is not None or r.consumption["code"] is not None), profile=prof,
                  verifier_version=v["verifier_version"])
    if again.to_ison() != r.to_ison():
        raise IGARError("RECEIPT_RECOMPUTE_MISMATCH", "document", "re-running the verification path under the named profile does not reproduce this receipt", "R6")

# isongraph-ext/src/isongraph_ext/igar/trust.py
"""Policy qualification (R4), delivery pins (R5) and the ONE operation that authorizes use of hints (25 §4).

Vocabulary, kept apart on purpose:
  well-formed  R0-R2   `check.inspect_receipt`   the receipt is a consistent claim
  bound        R3      `check.check_subject`     the claim is about exactly these bytes
  policy-fit   R4      `qualify_policy`          the claimed verifier / profile / baseline / posture is acceptable - says NOTHING about provenance
  pinned       R5      `DeliveryPins`            these receipt bytes came through the consumer's controlled, digest-pinned channel
  authorized   all     `authorize_hints`         pinned AND bound AND policy-fit AND passed, together, for the hints actually held
A rejection receipt can be well-formed, bound and pinned - delivered evidence of a rejection - and can never be authorized.
"""
from __future__ import annotations

import re
from dataclasses import dataclass, field

from ..igdm.records import DEFAULT_LIMITS, Limits
from ..igx.digest import sha256_hex
from ..strict_ison import BlockShape, StrictError, read_canonical
from .check import check_subject, inspect_receipt
from .errors import IGARError
from .receipt import Receipt, passed

POLICY_SPEC = "IGAR-POLICY/1"
_HEX64 = re.compile(r"[0-9a-f]{64}")
POLICY_SHAPES = (BlockShape("meta", "policy", ("policy_spec", "require_admission", "require_full_resolution"), True),
                 BlockShape("table", "verifiers", ("verifier", "verifier_version")), BlockShape("table", "core_commits", ("core_commit",)),
                 BlockShape("table", "profiles", ("profile_sha256",)), BlockShape("table", "runtime", ("runtime_ison_graph", "runtime_ison_py")))


@dataclass(frozen=True)
class TrustPolicy:
    verifiers: frozenset[tuple[str, str]]       # accepted (verifier, verifier_version); empty accepts none
    core_commits: frozenset[str]                # accepted baseline core_commit values; empty accepts none
    profiles: frozenset[str]                    # accepted verifier-profile digests (25 §2.4); empty accepts none
    runtime: frozenset[tuple[str, str]] = frozenset()   # accepted (runtime_ison_graph, runtime_ison_py); empty = do not constrain
    require_admission: bool = True
    require_full_resolution: bool = True

    @classmethod
    def from_ison(cls, data: bytes | str, *, limits: Limits = DEFAULT_LIMITS) -> "TrustPolicy":
        """The policy file (docs/igar-policy-format.md). Canonical ISON, byte budget first; an empty table accepts nothing."""
        def bad(field_: str, detail: str) -> IGARError:
            return IGARError("MALFORMED_POLICY", field_, detail, "R4")
        try:
            _text, b = read_canonical(data, POLICY_SHAPES, max_bytes=limits.max_payload_bytes, max_rows=limits.max_rows_per_table)
        except StrictError as e:
            raise bad(e.field, e.detail) from None
        m = b[("meta", "policy")][0]
        if m["policy_spec"] != POLICY_SPEC:
            raise bad("meta.policy.policy_spec", f"must be {POLICY_SPEC}")
        for c in ("require_admission", "require_full_resolution"):
            if m[c] not in ("true", "false"):
                raise bad(f"meta.policy.{c}", 'the string token "true" or "false"')
        def strs(table: str, *cols: str) -> list[tuple]:
            out = []
            for r in b[("table", table)]:
                if not all(isinstance(r[c], str) and r[c] for c in cols):
                    raise bad(f"table.{table}", "non-empty strings")
                out.append(tuple(r[c] for c in cols))
            if len(set(out)) != len(out):
                raise bad(f"table.{table}", "duplicate row")
            return out
        commits = [x[0] for x in strs("core_commits", "core_commit")]
        profiles = [x[0] for x in strs("profiles", "profile_sha256")]
        if not all(re.fullmatch(r"[0-9a-f]{40}", c) for c in commits) or not all(_HEX64.fullmatch(p) for p in profiles):
            raise bad("table", "core_commit is 40 and profile_sha256 is 64 lowercase hex")
        return cls(frozenset(strs("verifiers", "verifier", "verifier_version")), frozenset(commits), frozenset(profiles),
                   frozenset(strs("runtime", "runtime_ison_graph", "runtime_ison_py")), m["require_admission"] == "true", m["require_full_resolution"] == "true")


def qualify_policy(r: Receipt, policy: TrustPolicy) -> None:
    """R4: is the verifier this receipt CLAIMS acceptable to `policy`? Raises RECEIPT_POLICY_UNSATISFIED naming the first unsatisfied dimension.

    This is NOT a trust decision. It looks only at the receipt's own claims: it does not check delivery, it does not check any bytes,
    and a receipt that satisfies it may be forged. Use `authorize_hints` to authorize use. Order: verification, verifier, profile,
    core_commit, runtime, admission, resolution.
    """
    v = r.verifier

    def no(field_: str, detail: str) -> IGARError:
        return IGARError("RECEIPT_POLICY_UNSATISFIED", field_, detail, "R4")
    if not passed(r):
        raise no("verification", "a rejection receipt is evidence of rejection, never of approval")
    if (v["verifier"], v["verifier_version"]) not in policy.verifiers:
        raise no("verifier", f"{v['verifier']} {v['verifier_version']} is not an accepted verifier")
    if v["profile_sha256"] not in policy.profiles:
        raise no("profile", f"verifier profile {v['profile_sha256']} is not accepted")
    if v["core_commit"] not in policy.core_commits:
        raise no("core_commit", f"baseline {v['core_commit']} is not accepted")
    if policy.runtime and (v["runtime_ison_graph"], v["runtime_ison_py"]) not in policy.runtime:
        raise no("runtime", "runtime versions are not accepted")
    if policy.require_admission and r.result["admission"] != "admitted":
        raise no("admission", f"admission is {r.result['admission']}")
    if policy.require_full_resolution and (v["semantic_verification"], v["object_resolution"]) != ("full", "full"):
        raise no("resolution", "verification was not full")


@dataclass(frozen=True)
class DeliveryPins:
    """The consumer's own statement of which receipt files its controlled channel delivers: sha256 digests, e.g. from a vendoring pin file.

    This module cannot know HOW the pin file was protected; that is the consumer's boundary (25 §4.3). A pin set that an
    attacker can edit pins nothing.
    """
    receipts: frozenset[str]

    def __post_init__(self) -> None:
        if not all(isinstance(d, str) and _HEX64.fullmatch(d) for d in self.receipts):
            raise IGARError("MALFORMED_PINS", "receipts", "every pin is 64 lowercase hex", "R5")

    @classmethod
    def of(cls, *digests: str) -> "DeliveryPins":
        return cls(frozenset(digests))

    @classmethod
    def from_sha256sums(cls, text: bytes | str, *, names: frozenset[str]) -> "DeliveryPins":
        """Read `sha256sum` lines (`<64 hex><two spaces><name>`) and pin ONLY the files named in `names`; every name must be present exactly once."""
        data = text.decode("utf-8") if isinstance(text, (bytes, bytearray)) else text
        found: dict[str, str] = {}
        for line in data.splitlines():
            m = re.fullmatch(r"([0-9a-f]{64})  (\S.*)", line)
            if m is None:
                raise IGARError("MALFORMED_PINS", "sha256sums", f"not a sha256sum line: {line!r}", "R5")
            if m.group(2) in found and found[m.group(2)] != m.group(1):
                raise IGARError("MALFORMED_PINS", "sha256sums", f"{m.group(2)} is pinned twice with different digests", "R5")
            found[m.group(2)] = m.group(1)
        missing = sorted(names - set(found))
        if missing:
            raise IGARError("MALFORMED_PINS", "sha256sums", f"no pin for {missing}", "R5")
        return cls(frozenset(found[n] for n in names))


_MINT = object()


@dataclass(frozen=True)
class TrustedAdmission:
    """In-process evidence that `authorize_hints` established every condition for THESE bytes. Not serializable, not a file format.

    Constructed only by `authorize_hints`. (Same-process code can always forge a Python object; this is a guard against accident,
    not against a hostile caller in your own process.)
    """
    receipt_sha256: str
    target_artifact_sha256: str
    binding_sha256: str
    igdm_payload_sha256: str
    igdc_sha256: str
    profile_sha256: str
    verifier: str
    verifier_version: str
    core_commit: str
    _mint: object = field(default=None, repr=False, compare=False)

    def __post_init__(self) -> None:
        if self._mint is not _MINT:
            raise TypeError("TrustedAdmission is issued only by authorize_hints")


def _bytes(name: str, v: object) -> bytes:
    if not isinstance(v, (bytes, bytearray)):
        raise TypeError(f"{name} must be bytes, got {type(v).__name__}")
    return bytes(v)


def authorize_hints(receipt: bytes, *, pins: DeliveryPins, target: bytes, binding: bytes, payload: bytes, igdc: bytes, policy: TrustPolicy,
                    limits: Limits = DEFAULT_LIMITS) -> TrustedAdmission:
    """The ONLY operation that authorizes use of a hint set. All inputs are required keywords; `igdc` must be the hint document held.

    First failure wins, in this order (nothing unpinned is ever parsed): size budget, R5 delivery pin, R0-R2 well-formed and consistent,
    R3 subject (target, binding, payload AND the hint document), R4 policy (a rejection receipt always fails here). A receipt that names
    no hint document cannot authorize one. There is no partial success.
    """
    raw = _bytes("receipt", receipt)
    target, binding, payload, igdc = _bytes("target", target), _bytes("binding", binding), _bytes("payload", payload), _bytes("igdc", igdc)
    if len(raw) > limits.max_payload_bytes:
        raise IGARError("RECEIPT_LIMIT_EXCEEDED", "max_payload_bytes", f"{len(raw)} bytes > {limits.max_payload_bytes}", "R0")
    digest = sha256_hex(raw)
    if digest not in pins.receipts:                                        # R5 delivery
        raise IGARError("RECEIPT_NOT_PINNED", "receipt", "these receipt bytes are not in the consumer's pinned delivery set", "R5")
    r = inspect_receipt(raw, limits=limits)                                # R0-R2
    check_subject(r, target=target, binding=binding, payload=payload, igdc=igdc)   # R3
    qualify_policy(r, policy)                                              # R4 (passed, verifier, profile, baseline, runtime, admission, resolution)
    s, v = r.subject, r.verifier
    return TrustedAdmission(digest, s["target_artifact_sha256"], s["binding_sha256"], s["igdm_payload_sha256"], s["igdc_sha256"], v["profile_sha256"],
                            v["verifier"], v["verifier_version"], v["core_commit"], _MINT)

        return f"{sub}/{name}"

    def slug(i: str) -> str:
        return i.replace("/", "--")

    def row(id_, op, *, binding=None, payload=None, target=None, igdc=None, receipt=None, profile=None, policy=None, expect, code=None, step=None, detail=None):
        rows.append({"id": id_, "op": op, "binding": binding, "payload": payload, "target": target, "igdc": igdc, "receipt": receipt, "profile": profile, "policy": policy,
                     "expect": expect, "code": code, "step": step, "detail": detail})

    # ---- profiles -------------------------------------------------------------------------------------------------------------
    B_ALL = all_baselines()
    installed_retired = {**BASELINES, CORE: dataclasses.replace(BASELINES[CORE], retired=True)}     # a retired baseline whose Core runtime IS installed
    PROFILES = {
        "default": (DEFAULT_LIMITS, BASELINES, DEFAULT_REGISTRY, DEFAULT_POLICY),
        "all-baselines-allow-retired": (DEFAULT_LIMITS, B_ALL, DEFAULT_REGISTRY, AdmissionPolicy(allow_retired_baselines=True)),
        "all-baselines": (DEFAULT_LIMITS, B_ALL, DEFAULT_REGISTRY, DEFAULT_POLICY),
        "retired-installed": (DEFAULT_LIMITS, installed_retired, DEFAULT_REGISTRY, DEFAULT_POLICY),
        "group-depth-3": (Limits(max_group_depth=3), BASELINES, DEFAULT_REGISTRY, DEFAULT_POLICY),
        "payload-limit-16": (Limits(max_payload_bytes=16), BASELINES, DEFAULT_REGISTRY, DEFAULT_POLICY),
    }
    ptext = {n: reference_profile(*v) for n, v in PROFILES.items()}
    pfile = {n: put("profiles", f"{n}.igar-profile.ison", t) for n, t in ptext.items()}
    assert len({sha(t) for t in ptext.values()}) == len(ptext), "profiles must have distinct digests"

    # ---- inputs: copies of frozen igdm1 / igdc1 files -------------------------------------------------------------------------
    manifest = {r["id"]: r for r in loads((IGDM / "manifest.ison").read_text(encoding="utf-8")).blocks[1].rows}

    def inputs(src_id: str, binding_override: bytes | None = None, payload_override: bytes | None = None, target_override: bytes | None = None):
        m = manifest[src_id]
        t = target_override or (IGDM / "graphs" / m["target"]).read_bytes()
        b = binding_override or (IGDM / "bindings" / m["binding"]).read_bytes()
        p = payload_override or (IGDM / "payloads" / m["payload"]).read_bytes()
        return b, p, t

    def stage(name: str, b: bytes, p: bytes, t: bytes):
        return {"binding": put("inputs", f"{name}.igx.ison", b), "payload": put("inputs", f"{name}.igdm.ison", p), "target": put("inputs", f"{sha(t)[:16]}.ison", t)}

    HINTS_FILE = IGDC / "hints" / "ok--node-hints-full-row.igdc.ison"
    HINTS = HINTS_FILE.read_bytes()
    hints_path = put("inputs", "ok--node-hints-full-row.igdc.ison", HINTS)
    receipts: dict[str, str] = {}

    def issue_row(id_, src, profile="default", *, igdc=False, igdc_state=None, binding=None, payload=None, target=None, detail=None):
        b, p, t = inputs(src, binding, payload, target)
        n, lim, bases, reg, pol = profile, *PROFILES[profile]
        text = reference_receipt(b, p, t, profile_text=ptext[profile], limits=lim, baselines=bases, registry=reg, policy=pol,
                                 igdc_sha=sha(HINTS) if igdc and igdc_state is None else None, consumption=igdc_state or (("derived", None, None, None) if igdc else ("absent", None, None, None)))
        receipts[id_] = text
        row(id_, "issue", **stage(slug(id_), b, p, t), igdc=hints_path if igdc and igdc_state is None else None, receipt=put("receipts", f"{slug(id_)}.igar.ison", text),
            profile=pfile[profile], expect="ok", detail="; ".join(x for x in (f"verifier_version={VERSION}", "igdc" if igdc else None, detail) if x))
        return b, p, t

    FULL = "ok/node-hints-full-row"
    issue_row("ok/verified-admitted", FULL)
    issue_row("ok/verified-unrecognized-roles", "ok/x-roles-preserved")
    issue_row("ok/verified-admission-denied-stale", FULL, "retired-installed", detail="retired baseline whose Core runtime is installed: verification full, admission denied A1")
    LIM = "ok/limited-runtime-skips-resolution"
    issue_row("ok/verified-limited-resolution", LIM, "all-baselines-allow-retired", detail="retired baseline, runtime absent: limited / skipped, admitted by profile policy")
    base_b = ExtensionBinding.from_ison((IGDM / "bindings" / manifest[FULL]["binding"]).read_text(encoding="utf-8"))
    rc = B_ALL[LOSSY].receipts["python"].sha256
    lossy_b = dataclasses.replace(base_b, core_commit=LOSSY, producer_core_posture="LOSSY_KNOWN", core_conformance_receipt=rc, consumer=Consumer("python", "1.4.1", "LOSSY_KNOWN", rc))
    issue_row("ok/verified-consumer-lossy", FULL, "all-baselines", binding=lossy_b.to_ison().encode("utf-8"), detail="CORE_LOSSY posture with reason CONSUMER_POSTURE_LOSSY_KNOWN")
    issue_row("ok/rejected-target-mismatch", "reject/target-sha-mismatch")
    issue_row("ok/rejected-binding-malformed", FULL, binding=b"this is not ison", detail="V1: the subject still hashes the raw bytes, core_commit is null")
    issue_row("ok/rejected-payload-shape", "reject/noncanonical-text")
    issue_row("ok/rejected-payload-oversize", FULL, "payload-limit-16", detail="D1 budget preflight under the profile's limits")
    issue_row("ok/rejected-unresolved-object", "reject/unresolved-node")
    issue_row("ok/rejected-baseline-unknown", FULL, binding=dataclasses.replace(base_b, core_commit="4" * 40).to_ison().encode("utf-8"))
    issue_row("ok/with-igdc", FULL, igdc=True, detail="igdc_sha256 is the digest of vectors/igdc1 ok--node-hints-full-row (written independently)")
    issue_row("ok/igdc-resolution-required", LIM, "all-baselines-allow-retired", igdc=True, igdc_state=("failed", "RESOLUTION_REQUIRED", "P0", "object_resolution"))
    issue_row("ok/profile-non-default", FULL, "group-depth-3", detail="same inputs as ok/verified-admitted under another profile: a different receipt")

    R = lambda i: receipts[i]  # noqa: E731
    inp = {r["id"]: r for r in rows}                     # stage info per issue row
    good = inp["ok/verified-admitted"]

    # ---- inspect: R0-R2 -------------------------------------------------------------------------------------------------------
    def edit(base_id, fn) -> str:
        d = loads(R(base_id))
        fn(d)
        return dumps_canonical(d)

    blk = lambda d, n: next(b for b in d.blocks if b.name == n)  # noqa: E731

    def setv(block, **kv):
        def f(d):
            blk(d, block).rows[0].update(kv)
        return f

    def chain(*fns):
        def f(d):
            for fn in fns:
                fn(d)
        return f

    def inspect(id_, text, code, step, detail=None):
        data = text if isinstance(text, bytes) else text.encode("utf-8")
        row(id_, "inspect", receipt=put("receipts", f"{slug(id_)}.igar.ison", data), expect="ok" if code is None else "reject", code=code, step=step, detail=detail)

    A = "ok/verified-admitted"
    for rid in ("ok/verified-admitted", "ok/rejected-target-mismatch", "ok/verified-admission-denied-stale", "ok/with-igdc", "ok/igdc-resolution-required"):
        inspect(f"ok/inspect-{rid.removeprefix('ok/')}", R(rid), None, None)
    base = R(A)
    MR, IR = "MALFORMED_RECEIPT", "INCONSISTENT_RECEIPT"
    inspect("reject/receipt-noncanonical", base.replace("\n\n", "\n\n\n", 1), MR, "R0")
    inspect("reject/receipt-cr", base.replace("\n", "\r\n"), MR, "R0")
    inspect("reject/receipt-bom", "﻿" + base, MR, "R0")
    inspect("reject/receipt-not-utf8", b"\xff\xfe" + base.encode(), MR, "R0")
    inspect("reject/receipt-oversize", base, "RECEIPT_LIMIT_EXCEEDED", "R0", "limits:max_payload_bytes=64")
    inspect("reject/receipt-row-budget", R("ok/verified-unrecognized-roles"), "RECEIPT_LIMIT_EXCEEDED", "R1", "limits:max_rows_per_table=1")
    inspect("reject/receipt-unexpected-block", edit(A, lambda d: d.blocks.append(Block("table", "extra", ["a"], []))), MR, "R1")
    inspect("reject/receipt-missing-block", edit(A, lambda d: d.blocks.pop()), MR, "R1")
    inspect("reject/receipt-duplicate-block", edit(A, lambda d: d.blocks.append(d.blocks[0])), MR, "R1")
    inspect("reject/receipt-extra-field", edit(A, lambda d: (blk(d, "igar").fields.append("note"), blk(d, "igar").rows[0].update(note="x"))), MR, "R1")
    inspect("reject/receipt-embedded-digest-field", edit(A, lambda d: (blk(d, "igar").fields.append("receipt_sha256"), blk(d, "igar").rows[0].update(receipt_sha256="d" * 64))), MR, "R1")
    inspect("reject/receipt-bad-hex", edit(A, setv("subject", target_artifact_sha256="xyz")), MR, "R1")
    inspect("reject/receipt-profile-null", edit(A, setv("verifier", profile_sha256=None)), MR, "R1")
    inspect("reject/receipt-profile-bad-hex", edit(A, setv("verifier", profile_sha256="zz")), MR, "R1")
    inspect("reject/receipt-unknown-spec", edit(A, setv("igar", receipt_spec="IGAR/2")), MR, "R1")
    inspect("reject/receipt-status-outside-vocabulary", edit(A, setv("result", verification="maybe")), MR, "R1")
    inspect("reject/receipt-summary-row", base + "\n---\nrows 1", MR, "R1")
    inspect("reject/receipt-typed-column", base.replace("receipt_spec", "receipt_spec:int", 1), MR, "R1")
    inspect("reject/receipt-computed-column", base.replace("receipt_spec", "receipt_spec:computed", 1), MR, "R1")
    inspect("reject/receipt-reference-cell", base.replace("IGAR/1", ":igar:1", 1), MR, "R1")
    inspect("reject/receipt-meta-extra-row", edit(A, lambda d: blk(d, "failure").rows.append(dict(blk(d, "failure").rows[0]))), MR, "R1")
    for name in BLOCKS_R:
        def dup(d, name=name):
            b = blk(d, name.split(".")[1])
            b.fields.append(b.fields[0])
        inspect(f"reject/receipt-duplicate-column-{name.replace('.', '-')}", edit(A, dup), MR, "R1")
    inspect("reject/receipt-verified-but-failure-set", edit(A, setv("failure", check="D2", code="X", field="f")), IR, "R2")
    inspect("reject/receipt-rejected-but-failure-null", edit("ok/rejected-target-mismatch", setv("failure", check=None, code=None, field=None)), IR, "R2")
    inspect("reject/receipt-status-contradicts-check", edit("ok/rejected-target-mismatch", setv("result", verification="verified")), IR, "R2")
    inspect("reject/receipt-admitted-without-verification", edit("ok/rejected-target-mismatch", setv("result", admission="admitted")), IR, "R2")
    inspect("reject/receipt-admission-fields-without-admission", edit("ok/rejected-target-mismatch", setv("admission", receipt_posture="OK")), IR, "R2")
    inspect("reject/receipt-consumption-contradicts-digest", edit(A, setv("result", consumption="derived")), IR, "R2")
    inspect("reject/receipt-limited-without-flags", edit(A, setv("verifier", semantic_verification=None)), IR, "R2")
    inspect("reject/receipt-core-commit-null-without-v1", edit(A, setv("verifier", core_commit=None)), IR, "R2")

    # ---- subject: R3 ----------------------------------------------------------------------------------------------------------
    wd = inp["ok/with-igdc"]
    other_b, other_p, other_t = b"binding bytes", b"payload bytes", b"target bytes"

    def subject(id_, rid, ok=True, **over):
        src = inp[rid]
        kw = {k: src[k] for k in ("binding", "payload", "target")}
        kw["igdc"] = src["igdc"]
        kw.update(over)
        row(id_, "subject", receipt=src["receipt"], **kw, expect="ok" if ok else "reject", code=None if ok else "RECEIPT_SUBJECT_MISMATCH", step=None if ok else "R3")

    subject("ok/subject-exact", "ok/verified-admitted")
    subject("ok/subject-exact-with-igdc", "ok/with-igdc")
    subject("reject/subject-target-differs", "ok/verified-admitted", False, target=put("inputs", "other.target.ison", other_t))
    subject("reject/subject-binding-differs", "ok/verified-admitted", False, binding=put("inputs", "other.igx.ison", other_b))
    subject("reject/subject-payload-differs", "ok/verified-admitted", False, payload=put("inputs", "other.igdm.ison", other_p))
    subject("reject/subject-igdc-differs", "ok/with-igdc", False, igdc=put("inputs", "other.igdc.ison", b"other hints"))
    subject("reject/subject-igdc-unexpected", "ok/verified-admitted", False, igdc=hints_path)
    subject("reject/subject-igdc-missing", "ok/with-igdc", False, igdc=None)

    # ---- policy: R4 -----------------------------------------------------------------------------------------------------------
    d_default = sha(ptext["default"])
    gp = lambda **kw: policy_text(commits=(CORE,), profiles=(d_default,), **kw)  # noqa: E731
    POL = {
        "accept": gp(), "empty": policy_text(), "verifier-other": policy_text(verifiers=(("isongraph-ext", "other"),), commits=(CORE,), profiles=(d_default,)),
        "profile-other": policy_text(commits=(CORE,), profiles=("e" * 64,)), "commit-other": policy_text(commits=("f" * 40,), profiles=(d_default,)),
        "runtime-other": gp(runtime=(("0.0.1", "0.0.1"),)), "permissive": gp(admission=False, full=False),
    }
    polf = {n: put("policies", f"{n}.policy.ison", t) for n, t in POL.items()}

    def policy(id_, rid, pol, code=None, detail=None):
        row(id_, "policy", receipt=inp[rid]["receipt"], policy=polf[pol], expect="ok" if code is None else "reject", code=code, step=None if code is None else "R4", detail=detail)

    NO = "RECEIPT_POLICY_UNSATISFIED"
    policy("ok/policy-match", "ok/verified-admitted", "accept", detail="a policy match says nothing about provenance or bytes")
    policy("reject/policy-receipt-rejected", "ok/rejected-target-mismatch", "permissive", NO, "a rejection receipt never satisfies any policy: field verification")
    policy("reject/policy-verifier-not-allowed", "ok/verified-admitted", "verifier-other", NO)
    policy("reject/policy-profile-not-accepted", "ok/verified-admitted", "profile-other", NO)
    policy("reject/policy-core-commit-not-allowed", "ok/verified-admitted", "commit-other", NO)
    policy("reject/policy-runtime-not-allowed", "ok/verified-admitted", "runtime-other", NO)
    policy("reject/policy-admission-required", "ok/verified-admission-denied-stale", "accept", NO)
    policy("reject/policy-full-resolution-required", "ok/verified-limited-resolution", "accept", NO)
    policy("reject/policy-empty-policy", "ok/verified-admitted", "empty", NO, "the empty policy accepts nothing")

    # ---- authorize: pinned AND bound AND policy-fit AND passed, for the hints held --------------------------------------------
    def authorize(id_, rid, *, receipt=None, pins="receipt", pol="accept", code=None, step=None, detail=None, **over):
        src = inp[rid]
        kw = {k: src[k] for k in ("binding", "payload", "target")}
        kw["igdc"] = hints_path
        kw.update(over)
        row(id_, "authorize", receipt=receipt or src["receipt"], policy=polf[pol], **kw, expect="ok" if code is None else "reject", code=code, step=step,
            detail="; ".join(x for x in (f"pins:{pins}", detail) if x))

    authorize("ok/authorize-pinned-bound-fit", "ok/with-igdc", detail="pinned receipt, exact bytes, accepted profile and verifier")
    authorize("reject/authorize-unpinned-but-policy-fit", "ok/with-igdc", pins="other", code="RECEIPT_NOT_PINNED", step="R5",
              detail="a consistent, policy-satisfying receipt for the real bytes that nobody pinned (finding F1)")
    authorize("reject/authorize-unpinned-empty-pins", "ok/with-igdc", pins="empty", code="RECEIPT_NOT_PINNED", step="R5")
    authorize("reject/authorize-unpinned-garbage-never-parsed", "ok/with-igdc", receipt=put("receipts", "garbage.igar.ison", b"\xff\xfe not a receipt"), pins="other",
              code="RECEIPT_NOT_PINNED", step="R5", detail="nothing unpinned is parsed: R5 fires, not R0")
    authorize("reject/authorize-subject-target-differs", "ok/with-igdc", code="RECEIPT_SUBJECT_MISMATCH", step="R3", target=put("inputs", "other.target.ison", other_t))
    authorize("reject/authorize-subject-igdc-differs", "ok/with-igdc", code="RECEIPT_SUBJECT_MISMATCH", step="R3", igdc=put("inputs", "other.igdc.ison", b"other hints"))
    authorize("reject/authorize-receipt-names-no-igdc", "ok/verified-admitted", code="RECEIPT_SUBJECT_MISMATCH", step="R3", detail="a receipt without a hint digest cannot authorize hints")
    authorize("reject/authorize-pinned-rejection-receipt", "ok/rejected-target-mismatch", code="RECEIPT_SUBJECT_MISMATCH", step="R3",
              detail="delivered evidence of rejection: pinned, well-formed, and it names no hint document, so it can never authorize")
    authorize("reject/authorize-policy-verifier", "ok/with-igdc", pol="verifier-other", code=NO, step="R4")
    authorize("reject/authorize-policy-profile", "ok/with-igdc", pol="profile-other", code=NO, step="R4")
    authorize("reject/authorize-oversize", "ok/with-igdc", code="RECEIPT_LIMIT_EXCEEDED", step="R0", detail="limits:max_payload_bytes=64")

    # ---- recompute: R6 --------------------------------------------------------------------------------------------------------
    def recompute(id_, rid, *, profile="default", receipt=None, code=None, extra=None):
        src = inp[rid]
        row(id_, "recompute", receipt=receipt or src["receipt"], binding=src["binding"], payload=src["payload"], target=src["target"], igdc=None,
            profile=pfile[profile] if profile else None, expect="ok" if code is None else "reject", code=code, step=None if code is None else "R6",
            detail="; ".join(x for x in (f"verifier_version={VERSION}", extra) if x))

    recompute("ok/recompute-default-profile", "ok/verified-admitted")
    recompute("ok/recompute-non-default-profile", "ok/profile-non-default", profile="group-depth-3", extra="a historical receipt re-run under the profile it names")
    recompute("ok/recompute-with-igdc", "ok/with-igdc")
    recompute("reject/recompute-profile-absent", "ok/verified-admitted", profile=None, code="RECOMPUTE_UNAVAILABLE")
    recompute("reject/recompute-profile-differs", "ok/verified-admitted", profile="group-depth-3", code="RECOMPUTE_UNAVAILABLE")
    recompute("reject/recompute-historical-under-current-defaults", "ok/profile-non-default", profile="default", code="RECOMPUTE_UNAVAILABLE",
              extra="must NOT silently recompute under today's defaults")
    recompute("reject/recompute-verifier-version", "ok/verified-admitted", code="RECOMPUTE_UNAVAILABLE", extra="verifier_version=other")
    forged = edit("ok/verified-admitted", chain(setv("result", admission="denied"), setv("admission", code="STALE_BINDING", step="A1", receipt_posture="INADMISSIBLE"),
                                                 lambda d: blk(d, "reasons").rows.append({"reason": "BASELINE_RETIRED"})))
    recompute("reject/recompute-flipped-verdict", "ok/verified-admitted", receipt=put("receipts", "reject--recompute-flipped-verdict.igar.ison", forged), code="RECEIPT_RECOMPUTE_MISMATCH")
    wrong_base = edit("ok/verified-admitted", setv("verifier", core_commit="a" * 40))
    recompute("reject/recompute-wrong-baseline-field", "ok/verified-admitted", receipt=put("receipts", "reject--recompute-wrong-baseline-field.igar.ison", wrong_base), code="RECEIPT_RECOMPUTE_MISMATCH")
    bad_prof = ptext["default"] + "\n---\nx"
    bad_named = edit("ok/verified-admitted", setv("verifier", profile_sha256=sha(bad_prof)))
    row("reject/recompute-malformed-profile", "recompute", receipt=put("receipts", "reject--recompute-malformed-profile.igar.ison", bad_named), binding=good["binding"],
        payload=good["payload"], target=good["target"], profile=put("profiles", "malformed.igar-profile.ison", bad_prof), expect="reject", code="MALFORMED_PROFILE", step="R6",
        detail=f"verifier_version={VERSION}")

    # ---- profile documents ---------------------------------------------------------------------------------------------------
    for n in ("default", "all-baselines", "retired-installed"):
        row(f"ok/profile-{n}", "profile", profile=pfile[n], expect="ok")

    def pedit(fn) -> str:
        d = loads(ptext["default"])
        fn(d)
        return dumps_canonical(d)

    def bb(d, n):
        return next(b for b in d.blocks if b.name == n)

    MP = "MALFORMED_PROFILE"
    pcases = [
        ("wrong-spec", pedit(lambda d: bb(d, "profile").rows[0].update(profile_spec="IGAR-PROFILE/9")), MP, "R6", None),
        ("negative-limit", pedit(lambda d: bb(d, "limits").rows[0].update(max_group_depth=-1)), MP, "R6", None),
        ("bare-boolean-policy", pedit(lambda d: bb(d, "admission_policy").rows[0].update(allow_retired_baselines=True)), MP, "R6", None),
        ("duplicate-baseline", pedit(lambda d: bb(d, "baselines").rows.append(dict(bb(d, "baselines").rows[0]))), MP, "R6", None),
        ("receipt-for-unknown-baseline", pedit(lambda d: bb(d, "baseline_receipts").rows[0].update(baseline="8" * 40)), MP, "R6", None),
        ("unknown-port", pedit(lambda d: bb(d, "baseline_receipts").rows[0].update(port="cobol")), MP, "R6", None),
        ("missing-port-receipt", pedit(lambda d: bb(d, "baseline_receipts").rows.pop()), MP, "R6", None),
        ("bad-receipt-sha", pedit(lambda d: bb(d, "baseline_receipts").rows[0].update(sha256="xyz")), MP, "R6", None),
        ("duplicate-runtime-module", pedit(lambda d: bb(d, "baseline_runtime").rows.append(dict(bb(d, "baseline_runtime").rows[0]))), MP, "R6", None),
        ("unexpected-block", pedit(lambda d: d.blocks.append(Block("table", "extra", ["a"], []))), MP, "R6", None),
        ("duplicate-column-limits", pedit(lambda d: bb(d, "limits").fields.append("max_group_depth")), MP, "R6", None),
        ("summary-row", ptext["default"] + "\n---\nx", MP, "R6", None),
        ("not-utf8", b"\xff" + ptext["default"].encode(), MP, "R6", None),
        ("oversize", ptext["default"], "RECEIPT_LIMIT_EXCEEDED", "R6", "limits:max_payload_bytes=64"),
    ]
    for name, text, code, step, detail in pcases:
        row(f"reject/profile-{name}", "profile", profile=put("profiles", f"{name}.igar-profile.ison", text), expect="reject", code=code, step=step, detail=detail)

    # ---- policy files --------------------------------------------------------------------------------------------------------
    row("ok/policyfile-valid", "policyfile", policy=polf["accept"], expect="ok")
    row("ok/policyfile-empty-tables-accept-nothing", "policyfile", policy=polf["empty"], expect="ok", detail="legal, and accepts nothing")

    def poledit(fn) -> str:
        d = loads(POL["accept"])
        fn(d)
        return dumps_canonical(d)

    MPOL = "MALFORMED_POLICY"
    for name, text in [
        ("wrong-spec", policy_text(spec="IGAR-POLICY/9")), ("bare-boolean", poledit(lambda d: bb(d, "policy").rows[0].update(require_admission=True))),
        ("unexpected-block", poledit(lambda d: d.blocks.append(Block("table", "extra", ["a"], [])))),
        ("duplicate-column", poledit(lambda d: bb(d, "policy").fields.append("policy_spec"))),
        ("bad-commit-hex", policy_text(commits=("zz",), profiles=(d_default,))), ("bad-profile-hex", policy_text(commits=(CORE,), profiles=("zz",))),
        ("summary-row", POL["accept"] + "\n---\nx"), ("duplicate-row", policy_text(commits=(CORE, CORE), profiles=(d_default,))),
    ]:
        row(f"reject/policyfile-{name}", "policyfile", policy=put("policies", f"bad-{name}.policy.ison", text), expect="reject", code=MPOL, step="R4")

    # ---- manifest -------------------------------------------------------------------------------------------------------------
    cols = ["id", "op", "binding", "payload", "target", "igdc", "receipt", "profile", "policy", "expect", "code", "step", "detail"]
    for r in rows:
        assert set(r) == set(cols)
